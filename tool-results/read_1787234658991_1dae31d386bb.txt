    83→export default function CompanyPortal({ company, onLogout }: { company: any; onLogout: () => void }) {
    84→  const [tab, setTab] = useState<Tab>('profile')
    85→  const [data, setData] = useState<CompanyFull | null>(null)
    86→  const [customColumns, setCustomColumns] = useState<CustomColumn[]>([])
    87→  const [loading, setLoading] = useState(true)
    88→  const [revealed, setRevealed] = useState(false)
    89→  const { toast } = useToast()
    90→
    91→  const load = useCallback(async () => {
    92→    setLoading(true)
    93→    try {
    94→      const r = await api<{ company: CompanyFull; customColumns: CustomColumn[] }>('/api/company/self')
    95→      setData(r.company)
    96→      setCustomColumns(r.customColumns || [])
    97→    } catch (e: any) {
    98→      toast({ title: 'Error', description: e.message, variant: 'destructive' })
    99→    } finally {
   100→      setLoading(false)
   101→    }
   102→  }, [toast])
   103→
   104→  useEffect(() => {
   105→    load()
   106→  }, [load])
   107→
   108→  return (
   109→    <div className="min-h-screen flex flex-col bg-muted/20">
   110→      {/* Header */}
   111→      <header className="border-b bg-background sticky top-0 z-10">
   112→        <div className="max-w-5xl mx-auto px-4 sm:px-6 py-3 flex items-center justify-between">
   113→          <div className="flex items-center gap-3">
   114→            <div className="h-9 w-9 rounded-lg bg-primary text-primary-foreground grid place-items-center font-bold">
   115→              {company.name.charAt(0).toUpperCase()}
   116→            </div>
   117→            <div className="leading-tight">
   118→              <div className="font-semibold text-sm truncate max-w-[200px] sm:max-w-xs">{company.name}</div>
   119→              <div className="text-xs text-muted-foreground">Company Portal</div>
   120→            </div>
   121→          </div>
   122→          <div className="flex items-center gap-2">
   123→            <Button variant="ghost" size="sm" onClick={load} disabled={loading}>
   124→              {loading ? <Loader2 className="h-4 w-4 animate-spin" /> : <RefreshCw className="h-4 w-4" />}
   125→            </Button>
   126→            <Button variant="outline" size="sm" onClick={onLogout}>
   127→              <LogOut className="h-4 w-4 mr-1.5" /> Sign out
   128→            </Button>
   129→          </div>
   130→        </div>
   131→      </header>
   132→
   133→      {/* Isolation banner */}
   134→      <div className="bg-emerald-50 border-b border-emerald-200">
   135→        <div className="max-w-5xl mx-auto px-4 sm:px-6 py-2 flex items-center gap-2 text-xs text-emerald-800">
   136→          <ShieldCheck className="h-3.5 w-3.5 shrink-0" />
   137→          You are signed in as <strong>{company.name}</strong>. You can only view and edit your own company data.
   138→        </div>
   139→      </div>
   140→
   141→      {/* Tabs */}
   142→      <div className="border-b bg-background">
   143→        <div className="max-w-5xl mx-auto px-4 sm:px-6 flex gap-1 overflow-x-auto">
   144→          <TabBtn active={tab === 'profile'} onClick={() => setTab('profile')} icon={<Building2 className="h-4 w-4" />}>Profile</TabBtn>
   145→          <TabBtn active={tab === 'founders'} onClick={() => setTab('founders')} icon={<Users className="h-4 w-4" />}>Founders {data && <span className="ml-1 text-xs text-muted-foreground">({data.founders.length})</span>}</TabBtn>
   146→          <TabBtn active={tab === 'annual'} onClick={() => setTab('annual')} icon={<DollarSign className="h-4 w-4" />}>Annual Data {data && <span className="ml-1 text-xs text-muted-foreground">({data.annualData.length})</span>}</TabBtn>
   147→        </div>
   148→      </div>
   149→
   150→      <main className="flex-1 max-w-5xl w-full mx-auto px-4 sm:px-6 py-6">
   151→        {loading && !data ? (
   152→          <div className="py-12 text-center text-muted-foreground text-sm"><Loader2 className="h-5 w-5 animate-spin inline mr-2" />Loading your company data…</div>
   153→        ) : !data ? (
   154→          <div className="py-12 text-center text-muted-foreground text-sm">Failed to load.</div>
   155→        ) : (
   156→          <>
   157→            {tab === 'profile' && <ProfileTab company={data} customColumns={customColumns} onChanged={load} revealed={revealed} setRevealed={setRevealed} />}
   158→            {tab === 'founders' && <FoundersTab company={data} onChanged={load} />}
   159→            {tab === 'annual' && <AnnualTab company={data} onChanged={load} />}
   160→          </>

   162→      </main>

   164→      <footer className="border-t bg-background mt-auto">
   165→        <div className="max-w-5xl mx-auto px-4 sm:px-6 py-4 text-center text-xs text-muted-foreground">
   166→          SEECS Company Registry · NUST Islamabad

   190→function ProfileTab({ company, customColumns, onChanged, revealed, setRevealed }: { company: CompanyFull; customColumns: CustomColumn[]; onChanged: () => void; revealed: boolean; setRevealed: (b: boolean) => void }) {
   191→  const [name, setName] = useState(company.name)
   192→  const [description, setDescription] = useState(company.description || '')
   193→  const [email, setEmail] = useState(company.email || '')
   194→  const [phone, setPhone] = useState(company.phone || '')
   195→  const [website, setWebsite] = useState(company.website || '')
   196→  const [locationId, setLocationId] = useState(company.locationId || '')
   197→  const [customValues, setCustomValues] = useState<Record<string, string>>(() => {
   198→    const map: Record<string, string> = {}
   199→    company.customValues?.forEach((cv) => { map[cv.customColumnId] = cv.value })
   200→    return map
   201→  })
   202→  const [lookup, setLookup] = useState<Lookup | null>(null)
   203→  const [saving, setSaving] = useState(false)
   204→  const [savingCustom, setSavingCustom] = useState(false)
   205→  const { toast } = useToast()
   206→
   207→  useEffect(() => {
   208→    api<Lookup>('/api/company/lookup').then(setLookup).catch(() => {})
   209→  }, [])
   210→
   211→  const save = async (e: React.FormEvent) => {
   212→    e.preventDefault()
   213→    setSaving(true)
   214→    try {
   215→      await api('/api/company/self', { method: 'PUT', body: { name, description, email, phone, website, locationId: locationId || null } })
   216→      toast({ title: 'Saved', description: 'Your profile has been updated' })
   217→      onChanged()
   218→    } catch (e: any) {
   219→      toast({ title: 'Error', description: e.message, variant: 'destructive' })
   220→    } finally {
   221→      setSaving(false)
   222→    }
   223→  }
   224→
   225→  const saveCustomValues = async () => {
   226→ setSavingCustom(true)
   227→    try {
   228→      await api('/api/company/self', { method: 'PUT', body: { customValues } })
   229→      toast({ title: 'Saved', description: 'Custom fields updated' })
   230→      onChanged()
   231→    } catch (e: any) {
   232→      toast({ title: 'Error', description: e.message, variant: 'destructive' })
   233→    } finally {
   234→      setSavingCustom(false)
   235→    }
   236→  }
   237→
   238→  return (
   239→    <div className="space-y-4">
   240→      <Card>
   241→        <CardHeader>
   242→          <CardTitle className="text-base">Company profile</CardTitle>
   243→          <CardDescription>Update your company's public information. Sector and city are managed by the SEECS admin.</CardDescription>
   244→        </CardHeader>
   245→        <CardContent>
   246→          <form onSubmit={save} className="space-y-4">
   247→            <div className="grid sm:grid-cols-2 gap-3">
   248→              <div className="space-y-1.5 sm:col-span-2">
   249→                <Label>Company name</Label>
   250→                <Input value={name} onChange={(e) => setName(e.target.value)} required />
   251→              </div>
   252→              <div className="space-y-1.5 sm:col-span-2">
   253→                <Label>Description</Label>
   254→                <Textarea value={description} onChange={(e) => setDescription(e.target.value)} rows={3} />
   255→              </div>
   256→              <div className="space-y-1.5">
   257→                <Label>Email</Label>
   258→                <Input type="email" value={email} onChange={(e) => setEmail(e.target.value)} />
   259→              </div>
   260→              <div className="space-y-1.5">
   261→                <Label>Phone</Label>
   262→                <Input value={phone} onChange={(e) => setPhone(e.target.value)} />
   263→              </div>
   264→              <div className="space-y-1.5 sm:col-span-2">
   265→                <Label>Website</Label>
   266→                <Input value={website} onChange={(e) => setWebsite(e.target.value)} placeholder="https://…" />
   267→              </div>
   268→              <div className="space-y-1.5 sm:col-span-2">
   269→                <Label>Office location</Label>
   270→                <select
   271→                  className="flex h-9 w-full rounded-md border border-input bg-background px-3 py-1 text-sm ring-offset-background focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
   272→                  value={locationId}
   273→                  onChange={(e) => setLocationId(e.target.value)}
   274→                >
   275→                  <option value="">— None —</option>
   276→                  {lookup?.locations.map((l) => <option key={l.id} value={l.id}>{l.address} · {l.city.name}</option>)}
   277→                </select>
   278→              </div>
   279→            </div>
   280→
   281→            <div className="grid sm:grid-cols-2 gap-3 pt-3 border-t">
   282→              <div>
   283→                <Label className="text-xs text-muted-foreground">Sector (managed by admin)</Label>
   284→                <div className="text-sm font-medium mt-1">{company.sector.name}</div>
   285→              </div>
   286→              <div>
   287→                <Label className="text-xs text-muted-foreground">City (managed by admin)</Label>
   288→                <div className="text-sm font-medium mt-1">{company.city.name}</div>
   289→              </div>
   290→            </div>
   291→
   292→            {(company.status || company.foundedYear || company.sinceDate || company.branchesCount || company.revenue != null || company.revenueMin != null) && (
   293→              <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-3 pt-3 border-t">
   294→                {company.status && (
   295→                  <div>
   296→                    <Label className="text-xs text-muted-foreground">Status</Label>
   297→                    <div className="mt-1">
   298→                      <Badge variant="outline" className={`text-xs ${STATUS_STYLES[company.status] || ''}`}>{company.status}</Badge>
   299→                    </div>
   300→                  </div>
   301→                )}
   302→                {company.foundedYear && (
   303→                  <div>
   304→                    <Label className="text-xs text-muted-foreground">Founded year</Label>
   305→                    <div className="text-sm font-medium mt-1">{company.foundedYear}</div>
   306→                  </div>
   307→                )}
   308→                {company.sinceDate && (
   309→                  <div>
   310→                    <Label className="text-xs text-muted-foreground">Since date</Label>
   311→                    <div className="text-sm font-medium mt-1">{company.sinceDate}</div>
   312→                  </div>
   313→                )}
   314→                {company.branchesCount != null && (
   315→                  <div>
   316→                    <Label className="text-xs text-muted-foreground">Branches</Label>
   317→                    <div className="text-sm font-medium mt-1">{company.branchesCount}</div>
   318→                  </div>
   319→                )}
   320→                {company.revenue != null ? (
   321→                  <div>
   322→                    <Label className="text-xs text-muted-foreground">Revenue</Label>
   323→                    <div className="text-sm font-medium mt-1">{formatCurrency(company.revenue)}</div>
   324→                  </div>
   325→                ) : (company.revenueMin != null || company.revenueMax != null) ? (
   326→                  <div>
   327→                    <Label className="text-xs text-muted-foreground">Revenue range</Label>
   328→                    <div className="text-sm font-medium mt-1">{formatCurrency(company.revenueMin || 0)} – {formatCurrency(company.revenueMax || 0)}</div>
   329→                  </div>
   330→                ) : null}
   331→                {company.statusReason && (
   332→                  <div className="sm:col-span-2">
   333→                    <Label className="text-xs text-muted-foreground">Status reason</Label>
   334→                    <div className="text-sm font-medium mt-1">{company.statusReason}</div>
   335→                  </div>
   336→                )}
   337→              </div>
   338→            )}
   339→
   340→            <div className="flex justify-end">
   341→              <Button type="submit" disabled={saving}>
   342→                {saving && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   343→                Save changes
   344→              </Button>
   345→            </div>
   346→          </form>
   347→        </CardContent>
   348→      </Card>
   349→
   350→      {/* Custom columns card */}
   351→      {customColumns.length > 0 && (
   352→        <Card>
   353→          <CardHeader>
   354→            <CardTitle className="text-base flex items-center gap-2"><Columns3 className="h-4 w-4" /> Custom fields</CardTitle>
   355→            <CardDescription>Additional fields defined by the SEECS admin. Fill these in to provide extra information about your company.</CardDescription>
   356→          </CardHeader>

   358→            <div className="grid sm:grid-cols-2 gap-3">
   359→              {customColumns.map((col) => (
   360→                <div key={col.id} className="space-y-1.5">
   361→                  <Label>{col.name}{col.required && <span className="text-destructive ml-1">*</span>}{col.description && <span className="text-xs text-muted-foreground ml-1.5">— {col.description}</span>}</Label>

   363→                    <Input

   371→                      type="date"
   372→                      value={customValues[col.id] || ''}
   373→                      onChange={(e) => setCustomValues((s) => ({ ...s, [col.id]: e.target.value }))}

   386→                    <Input
   387→                      type="url"
   388→                      value={customValues[col.id] || ''}
   389→                      onChange={(e) => setCustomValues((s) => ({ ...s, [col.id]: e.target.value }))}
   390→                      placeholder="https://…"
   391→                    />
   392→                  ) : (
   393→                    <Input
   394→                      value={customValues[col.id] || ''}
   395→                      onChange={(e) => setCustomValues((s) => ({ ...s, [col.id]: e.target.value }))}
   396→                      placeholder={col.required ? 'Required' : 'Optional'}
   397→                    />
   398→                  )}
   399→                </div>
   400→              ))}
   401→            </div>
   402→            <div className="flex justify-end">
   403→              <Button onClick={saveCustomValues} disabled={savingCustom}>
   404→                {savingCustom && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   405→                Save custom fields
   406→              </Button>
   407→            </div>
   408→          </CardContent>
   409→        </Card>
   410→      )}
   411→
   412→      {/* API key card */}
   413→      <Card>
   414→        <CardHeader>
   415→          <CardTitle className="text-base flex items-center gap-2"><KeyRound className="h-4 w-4" /> Your API key</CardTitle>
   416→          <CardDescription>Keep this secret. Anyone with it can sign in to your company portal.</CardDescription>
   417→        </CardHeader>
   418→        <CardContent className="space-y-3">
   419→          <div className="flex items-center gap-2">
   420→            <code className="flex-1 text-xs font-mono bg-muted px-3 py-2.5 rounded break-all">
   421→              {revealed ? company.apiKey : `${company.apiKey.slice(0, 14)}••••••••••••••••`}
   422→            </code>
   423→            <Button variant="outline" size="sm" onClick={() => setRevealed(!revealed)}>
   424→              {revealed ? 'Hide' : 'Reveal'}
   425→            </Button>
   426→            <Button variant="outline" size="sm" onClick={() => { navigator.clipboard?.writeText(company.apiKey); toast({ title: 'Copied' }) }}>
   427→              <Copy className="h-4 w-4" />
   428→            </Button>
   429→          </div>
   430→          <Alert>
   431→            <AlertDescription>
   432→              To rotate or revoke this key, contact the SEECS database manager. A revoked key immediately loses access.
   433→            </AlertDescription>
   434→          </Alert>
   435→        </CardContent>
   436→      </Card>
   437→    </div>
   438→  )
   439→}

   771→function AnnualFormDialog({ open, onOpenChange, existing, onSaved }: { open: boolean; onOpenChange: (o: boolean) => void; existing?: { id: string; year: number; monthlyRevenue: number; totalRevenue: number; employeeCount: number; projectCount: number; notes: string }; onSaved: () => void }) {
   772→  const [year, setYear] = useState('')
   773→  const [monthlyRevenue, setMonthlyRevenue] = useState('')
   774→  const [totalRevenue, setTotalRevenue] = useState('')
   775→  const [employeeCount, setEmployeeCount] = useState('')
   776→  const [projectCount, setProjectCount] = useState('')
   777→  const [notes, setNotes] = useState('')
   778→  const [saving, setSaving] = useState(false)
   779→  const { toast } = useToast()
   780→
   781→  useEffect(() => {
   782→    if (open) {
   783→      setYear(existing ? String(existing.year) : String(new Date().getFullYear() - 1))
   784→      setMonthlyRevenue(existing ? String(existing.monthlyRevenue) : '')
   785→      setTotalRevenue(existing ? String(existing.totalRevenue) : '')
   786→      setEmployeeCount(existing ? String(existing.employeeCount) : '')
   787→      setProjectCount(existing ? String(existing.projectCount) : '')
   788→      setNotes(existing?.notes || '')
   789→    }
   790→  }, [open, existing])
   791→
   792→  const submit = async (e: React.FormEvent) => {
   793→    e.preventDefault()
   794→    setSaving(true)
   795→    try {
   796→      const body = {
   797→        year: Number(year),
   798→        monthlyRevenue: Number(monthlyRevenue) || 0,
   799→        totalRevenue: Number(totalRevenue) || (Number(monthlyRevenue) || 0) * 12,
   800→        employeeCount: Number(employeeCount) || 0,
   801→        projectCount: Number(projectCount) || 0,
   802→        notes: notes || null,
   803→      }
   804→      if (existing) {
   805→        await api(`/api/company/self/annual-data/${existing.id}`, { method: 'PUT', body })
   806→        toast({ title: 'Saved', description: 'Annual data updated' })

   808→        await api('/api/company/self/annual-data', { method: 'POST', body })
   809→        toast({ title: 'Added', description: 'Annual data added' })
   810→      }
   811→      onSaved()
   812→    } catch (e: any) {
   813→      toast({ title: 'Error', description: e.message, variant: 'destructive' })
   814→    } finally {
   815→      setSaving(false)
   816→    }
   817→  }
   818→
   819→  return (
   820→    <Dialog open={open} onOpenChange={onOpenChange}>

   823→          <DialogTitle>{existing ? 'Edit annual data' : 'Add annual data'}</DialogTitle>

   833→              <Label>Monthly revenue (PKR)</Label>
