    91→export default function CompanyPortal({ company, onLogout }: { company: any; onLogout: () => void }) {
    92→  const [tab, setTab] = useState<Tab>('profile')
    93→  const [data, setData] = useState<CompanyFull | null>(null)
    94→  const [customColumns, setCustomColumns] = useState<CustomColumn[]>([])

    96→  const [revealed, setRevealed] = useState(false)
    97→  const { toast } = useToast()
    98→
    99→  const load = useCallback(async () => {
   100→    setLoading(true)
   101→    try {
   102→      const r = await api<{ company: CompanyFull; customColumns: CustomColumn[] }>('/api/company/self')
   103→      setData(r.company)
   104→      setCustomColumns(r.customColumns || [])
   105→    } catch (e: any) {
   106→      toast({ title: 'Error', description: e.message, variant: 'destructive' })
   107→    } finally {
   108→      setLoading(false)
   109→    }
   110→  }, [toast])
   111→
   112→  useEffect(() => {
   113→    load()
   114→  }, [load])
   115→
   116→  const statusBadge = data?.status ? (
   117→    <Badge variant="outline" className={`text-xs font-medium ${STATUS_STYLES[data.status] || ''}`}>
   118→      {data.status === 'Active' && <span className="inline-block h-1.5 w-1.5 rounded-full bg-emerald-500 mr-1.5" />}
   119→      {data.status === 'Inactive' && <span className="inline-block h-1.5 w-1.5 rounded-full bg-amber-500 mr-1.5" />}
   120→      {data.status === 'Discontinued' && <span className="inline-block h-1.5 w-1.5 rounded-full bg-rose-500 mr-1.5" />}
   121→      {data.status}
   122→    </Badge>
   123→  ) : null
   124→
   125→  const partialKey = data?.apiKey ? `${data.apiKey.slice(0, 10)}${'•'.repeat(8)}` : ''
   126→
   127→  return (
   128→    <div className="min-h-screen flex flex-col bg-muted/20">
   129→      {/* Professional Header Bar */}
   130→      <header className="border-b bg-background sticky top-0 z-10 shadow-sm">
   131→        <div className="max-w-5xl mx-auto px-4 sm:px-6 py-3 flex items-center justify-between">
   132→          <div className="flex items-center gap-3">
   133→            <div className="h-10 w-10 rounded-xl bg-gradient-to-br from-emerald-500 to-teal-600 text-white grid place-items-center font-bold text-sm shadow-md">
   134→              {company.name.charAt(0).toUpperCase()}
   135→            </div>
   136→            <div className="leading-tight">
   137→              <div className="flex items-center gap-2">
   138→                <span className="font-semibold text-sm truncate max-w-[180px] sm:max-w-xs">{data?.name || company.name}</span>
   139→                {statusBadge}
   140→              </div>
   141→              <div className="flex items-center gap-1.5 mt-0.5">
   142→                <KeyRound className="h-3 w-3 text-muted-foreground" />
   143→                <code className="text-[11px] font-mono text-muted-foreground">{partialKey}</code>
   144→              </div>
   145→            </div>
   146→          </div>
   147→          <div className="flex items-center gap-2">
   148→            <ThemeToggle />
   149→            <Button variant="ghost" size="sm" onClick={load} disabled={loading}>
   150→              {loading ? <Loader2 className="h-4 w-4 animate-spin" /> : <RefreshCw className="h-4 w-4" />}
   151→            </Button>
   152→            <Button variant="outline" size="sm" onClick={onLogout}>
   153→              <LogOut className="h-4 w-4 mr-1.5" /> Sign out
   154→            </Button>
   155→          </div>
   156→        </div>
   157→      </header>
   158→
   159→      {/* Isolation banner */}
   160→      <div className="bg-emerald-50 dark:bg-emerald-950 border-b border-emerald-200 dark:border-emerald-800">
   161→        <div className="max-w-5xl mx-auto px-4 sm:px-6 py-2 flex items-center gap-2 text-xs text-emerald-800 dark:text-emerald-200">
   162→          <ShieldCheck className="h-3.5 w-3.5 shrink-0" />
   163→          You are signed in as <strong>{company.name}</strong>. You can only view and edit your own company data.
   164→        </div>
   165→      </div>
   166→
   167→      {/* Tabs */}
   168→      <div className="border-b bg-background">
   169→        <div className="max-w-5xl mx-auto px-4 sm:px-6 flex gap-1 overflow-x-auto">
   170→          <TabBtn active={tab === 'profile'} onClick={() => setTab('profile')} icon={<Building2 className="h-4 w-4" />}>Profile</TabBtn>
   171→          <TabBtn active={tab === 'founders'} onClick={() => setTab('founders')} icon={<Users className="h-4 w-4" />}>Founders {data && <span className="ml-1 text-xs text-muted-foreground">({data.founders.length})</span>}</TabBtn>
   172→          <TabBtn active={tab === 'annual'} onClick={() => setTab('annual')} icon={<DollarSign className="h-4 w-4" />}>Annual Data {data && <span className="ml-1 text-xs text-muted-foreground">({data.annualData.length})</span>}</TabBtn>
   173→        </div>
   174→      </div>
   175→
   176→      <main className="flex-1 max-w-5xl w-full mx-auto px-4 sm:px-6 py-6">
   177→        {loading && !data ? (
   178→          <div className="py-12 text-center text-muted-foreground text-sm"><Loader2 className="h-5 w-5 animate-spin inline mr-2" />Loading your company data…</div>
   179→        ) : !data ? (
   180→          <div className="py-12 text-center text-muted-foreground text-sm">Failed to load.</div>
   181→        ) : (
   182→          <>
   183→            {tab === 'profile' && <ProfileTab company={data} customColumns={customColumns} onChanged={load} revealed={revealed} setRevealed={setRevealed} />}
   184→            {tab === 'founders' && <FoundersTab company={data} onChanged={load} />}
   185→            {tab === 'annual' && <AnnualTab company={data} onChanged={load} />}

   187→        )}
   188→      </main>
   189→
   190→      <footer className="border-t bg-background mt-auto">
   191→        <div className="max-w-5xl mx-auto px-4 sm:px-6 py-4 text-center text-xs text-muted-foreground">
   192→          SEECS Company Registry · NUST Islamabad
   193→        </div>
   194→      </footer>
   195→    </div>
   196→  )
   197→}

   216→function QuickStats({ company }: { company: CompanyFull }) {

   218→    ? company.annualData.reduce((a, b) => (a.year > b.year ? a : b))
   219→    : null

   224→      value: company.founders.length,
   225→      icon: <UserRound className="h-4 w-4" />,

   228→      bgLight: 'bg-emerald-50',
   229→    },

   238→    {
   239→      label: 'Latest Revenue',

   249→    <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">

   252→          ` transition-shadow hover:shadow-md`}>

   258→              <div className="text-xs text-muted-foreground font-medium">{s.label}</div>

   263→        </div>
   264→      ))}
   265→    </div>
   266→  )
   267→}

   287→function ProfileTab({ company, customColumns, onChanged, revealed, setRevealed }: { company: CompanyFull; customColumns: CustomColumn[]; onChanged: () => void; revealed: boolean; setRevealed: (b: boolean) => void }) {
   288→  const [name, setName] = useState(company.name)
   289→  const [description, setDescription] = useState(company.description || '')
   290→  const [email, setEmail] = useState(company.email || '')
   291→  const [phone, setPhone] = useState(company.phone || '')
   292→  const [website, setWebsite] = useState(company.website || '')
   293→  const [locationId, setLocationId] = useState(company.locationId || '')
   294→  const [customValues, setCustomValues] = useState<Record<string, string>>(() => {
   295→    const map: Record<string, string> = {}
   296→    company.customValues?.forEach((cv) => { map[cv.customColumnId] = cv.value })
   297→    return map
   298→  })
   299→  const [lookup, setLookup] = useState<Lookup | null>(null)
   300→  const [saving, setSaving] = useState(false)
   301→  const [savingCustom, setSavingCustom] = useState(false)
   302→  const { toast } = useToast()
   303→
   304→  useEffect(() => {
   305→    api<Lookup>('/api/company/lookup').then(setLookup).catch(() => {})
   306→  }, [])
   307→
   308→  const save = async (e: React.FormEvent) => {
   309→    e.preventDefault()
   310→    setSaving(true)
   311→    try {
   312→      await api('/api/company/self', { method: 'PUT', body: { name, description, email, phone, website, locationId: locationId || null } })
   313→      toast({ title: 'Saved', description: 'Your profile has been updated' })
   314→      onChanged()
   315→    } catch (e: any) {
   316→      toast({ title: 'Error', description: e.message, variant: 'destructive' })
   317→    } finally {
   318→      setSaving(false)
   319→    }
   320→  }
   321→
   322→  const saveCustomValues = async () => {
   323→    setSavingCustom(true)
   324→    try {
   325→      await api('/api/company/self', { method: 'PUT', body: { customValues } })
   326→      toast({ title: 'Saved', description: 'Custom fields updated' })
   327→      onChanged()
   328→    } catch (e: any) {
   329→      toast({ title: 'Error', description: e.message, variant: 'destructive' })
   330→    } finally {
   331→      setSavingCustom(false)
   332→    }
   333→  }
   334→
   335→  return (
   336→    <div className="space-y-5">
   337→      {/* Quick Stats */}
   338→      <QuickStats company={company} />
   339→
   340→      <Card>
   341→        <CardHeader>
   342→          <CardTitle className="text-base">Company profile</CardTitle>
   343→          <CardDescription>Update your company's public information. Sector and city are managed by the SEECS admin.</CardDescription>
   344→        </CardHeader>
   345→        <CardContent>
   346→          <form onSubmit={save} className="space-y-4">
   347→            <div className="grid sm:grid-cols-2 gap-3">
   348→              <div className="space-y-1.5 sm:col-span-2">
   349→                <Label>Company name</Label>
   350→                <Input value={name} onChange={(e) => setName(e.target.value)} required />
   351→              </div>
   352→              <div className="space-y-1.5 sm:col-span-2">
   353→                <Label>Description</Label>
   354→                <Textarea value={description} onChange={(e) => setDescription(e.target.value)} rows={3} />
   355→              </div>
   356→              <div className="space-y-1.5">
   357→                <Label>Email</Label>
   358→                <Input type="email" value={email} onChange={(e) => setEmail(e.target.value)} />
   359→              </div>
   360→              <div className="space-y-1.5">
   361→                <Label>Phone</Label>
   362→                <Input value={phone} onChange={(e) => setPhone(e.target.value)} />
   363→              </div>
   364→              <div className="space-y-1.5 sm:col-span-2">
   365→                <Label>Website</Label>
   366→                <Input value={website} onChange={(e) => setWebsite(e.target.value)} placeholder="https://…" />
   367→              </div>
   368→              <div className="space-y-1.5 sm:col-span-2">
   369→                <Label>Office location</Label>
   370→                <select
   371→                  className="flex h-9 w-full rounded-md border border-input bg-background px-3 py-1 text-sm ring-offset-background focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
   372→                  value={locationId}
   373→                  onChange={(e) => setLocationId(e.target.value)}
   374→                >
   375→                  <option value="">— None —</option>
   376→                  {lookup?.locations.map((l) => <option key={l.id} value={l.id}>{l.address} · {l.city.name}</option>)}
   377→                </select>
   378→              </div>
   379→            </div>
   380→
   381→            <div className="grid sm:grid-cols-2 gap-3 pt-3 border-t">
   382→              <div>
   383→                <Label className="text-xs text-muted-foreground">Sector (managed by admin)</Label>
   384→                <div className="text-sm font-medium mt-1">{company.sector.name}</div>
   385→              </div>
   386→              <div>
   387→                <Label className="text-xs text-muted-foreground">City (managed by admin)</Label>
   388→                <div className="text-sm font-medium mt-1">{company.city.name}</div>
   389→              </div>
   390→            </div>
   391→
   392→            {(company.status || company.foundedYear || company.sinceDate || company.branchesCount || company.revenue != null || company.revenueMin != null) && (
   393→              <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-3 pt-3 border-t">
   394→                {company.status && (
   395→                  <div>
   396→                    <Label className="text-xs text-muted-foreground">Status</Label>
   397→                    <div className="mt-1">
   398→                      <Badge variant="outline" className={`text-xs ${STATUS_STYLES[company.status] || ''}`}>{company.status}</Badge>
   399→                    </div>
   400→                  </div>
   401→                )}
   402→                {company.foundedYear && (
   403→                  <div>
   404→                    <Label className="text-xs text-muted-foreground">Founded year</Label>
   405→                    <div className="text-sm font-medium mt-1">{company.foundedYear}</div>
   406→                  </div>
   407→                )}
   408→                {company.sinceDate && (
   409→                  <div>
   410→                    <Label className="text-xs text-muted-foreground">Since date</Label>
   411→                    <div className="text-sm font-medium mt-1">{company.sinceDate}</div>
   412→                  </div>
   413→                )}
   414→                {company.branchesCount != null && (
   415→                  <div>
   416→                    <Label className="text-xs text-muted-foreground">Branches</Label>
   417→                    <div className="text-sm font-medium mt-1">{company.branchesCount}</div>
   418→                  </div>
   419→                )}
   420→                {company.revenue != null ? (
   421→                  <div>
   422→                    <Label className="text-xs text-muted-foreground">Revenue</Label>
   423→                    <div className="text-sm font-medium mt-1">{formatCurrency(company.revenue)}</div>
   424→                  </div>
   425→                ) : (company.revenueMin != null || company.revenueMax != null) ? (
   426→                  <div>
   427→                    <Label className="text-xs text-muted-foreground">Revenue range</Label>
   428→                    <div className="text-sm font-medium mt-1">{formatCurrency(company.revenueMin || 0)} – {formatCurrency(company.revenueMax || 0)}</div>
   429→                  </div>
   430→                ) : null}
   431→                {company.statusReason && (
   432→                  <div className="sm:col-span-2">
   433→                    <Label className="text-xs text-muted-foreground">Status reason</Label>
   434→                    <div className="text-sm font-medium mt-1">{company.statusReason}</div>
   435→                  </div>
   436→                )}
   437→              </div>
   438→            )}
   439→
   440→            <div className="flex justify-end">
   441→              <Button type="submit" disabled={saving}>
   442→                {saving && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   443→                Save changes
   444→              </Button>
   445→            </div>
   446→          </form>
   447→        </CardContent>
   448→      </Card>
   449→
   450→      {/* Custom columns card — dashed border with info icon */}
   451→      {customColumns.length > 0 && (
   452→        <div className="rounded-xl border-2 border-dashed border-muted-foreground/25 bg-muted/20 p-5">
   453→          <div className="flex items-center gap-2 mb-1">
   454→            <Columns3 className="h-4 w-4 text-muted-foreground" />
   455→            <h3 className="text-base font-semibold">Custom fields</h3>
   456→            <div className="ml-auto">
   457→              <Info className="h-4 w-4 text-muted-foreground" />
   458→            </div>
   459→          </div>
   460→          <p className="text-xs text-muted-foreground mb-4">
   461→            Additional fields defined by the SEECS admin. Fill these in to provide extra information about your company.
   462→          </p>
   463→          <div className="grid sm:grid-cols-2 gap-3">

   892→function AnnualFormDialog({ open, onOpenChange, existing, onSaved }: { open: boolean; onOpenChange: (o: boolean) => void; existing?: { id: string; year: number; monthlyRevenue: number; totalRevenue: number; employeeCount: number; projectCount: number; notes: string }; onSaved: () => void }) {

   894→  const [monthlyRevenue, setMonthlyRevenue] = useState('')
   895→  const [totalRevenue, setTotalRevenue] = useState('')
   896→  const [employeeCount, setEmployeeCount] = useState('')
   897→  const [projectCount, setProjectCount] = useState('')
   898→  const [notes, setNotes] = useState('')
   899→  const [saving, setSaving] = useState(false)
   900→  const { toast } = useToast()
   901→
   902→  useEffect(() => {
   903→    if (open) {
   904→      setYear(existing ? String(existing.year) : String(new Date().getFullYear() - 1))
   905→      setMonthlyRevenue(existing ? String(existing.monthlyRevenue) : '')
   906→      setTotalRevenue(existing ? String(existing.totalRevenue) : '')
   907→      setEmployeeCount(existing ? String(existing.employeeCount) : '')
   908→      setProjectCount(existing ? String(existing.projectCount) : '')
   909→      setNotes(existing?.notes || '')
   910→    }
   911→  }, [open, existing])
   912→
   913→  const submit = async (e: React.FormEvent) => {
   914→    e.preventDefault()
   915→    setSaving(true)
   916→    try {
   917→      const body = {
   918→        year: Number(year),
   919→        monthlyRevenue: Number(monthlyRevenue) || 0,
   920→        totalRevenue: Number(totalRevenue) || (Number(monthlyRevenue) || 0) * 12,
   921→        employeeCount: Number(employeeCount) || 0,
   922→        projectCount: Number(projectCount) || 0,
   923→        notes: notes || null,
   924→      }
   925→      if (existing) {
   926→        await api(`/api/company/self/annual-data/${existing.id}`, { method: 'PUT', body })
   927→        toast({ title: 'Saved', description: 'Annual data updated' })
   928→      } else {
   929→        await api('/api/company/self/annual-data', { method: 'POST', body })
   930→        toast({ title: 'Added', description: 'Annual data added' })
   931→      }
