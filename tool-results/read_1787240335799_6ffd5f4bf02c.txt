   106→    } finally {
   107→      setLoading(false)
   108→    }
   109→  }, [toast])
   110→
   111→  useEffect(() => {
   112→    load()
   113→  }, [load])
   114→
   115→  const statusBadge = data?.status ? (
   116→    <Badge variant="outline" className={`text-xs font-medium ${STATUS_STYLES[data.status] || ''}`}>
   117→      {data.status === 'Active' && <span className="inline-block h-1.5 w-1.5 rounded-full bg-emerald-500 mr-1.5" />}
   118→      {data.status === 'Inactive' && <span className="inline-block h-1.5 w-1.5 rounded-full bg-amber-500 mr-1.5" />}
   119→      {data.status === 'Discontinued' && <span className="inline-block h-1.5 w-1.5 rounded-full bg-rose-500 mr-1.5" />}
   120→      {data.status}
   121→    </Badge>
   122→  ) : null
   123→
   124→  const partialKey = data?.apiKey ? `${data.apiKey.slice(0, 10)}${'•'.repeat(8)}` : ''

   126→  return (
   127→    <div className="min-h-screen flex flex-col bg-muted/20">
   128→      {/* Professional Header Bar */}
   129→      <header className="border-b bg-background sticky top-0 z-10 shadow-sm">
   130→        <div className="max-w-5xl mx-auto px-4 sm:px-6 py-3 flex items-center justify-between">
   131→          <div className="flex items-center gap-3">
   132→            <div className="h-10 w-10 rounded-xl bg-gradient-to-br from-emerald-500 to-teal-600 text-white grid place-items-center font-bold text-sm shadow-md">
   133→              {company.name.charAt(0).toUpperCase()}
   134→            </div>
   135→            <div className="leading-tight">
   136→              <div className="flex items-center gap-2">
   137→                <span className="font-semibold text-sm truncate max-w-[180px] sm:max-w-xs">{data?.name || company.name}</span>
   138→                {statusBadge}
   139→              </div>
   140→              <div className="flex items-center gap-1.5 mt-0.5">
   141→                <KeyRound className="h-3 w-3 text-muted-foreground" />
   142→                <code className="text-[11px] font-mono text-muted-foreground">{partialKey}</code>
   143→              </div>
   144→            </div>
   145→          </div>
   146→          <div className="flex items-center gap-2">
   147→            <Button variant="ghost" size="sm" onClick={load} disabled={loading}>
   148→              {loading ? <Loader2 className="h-4 w-4 animate-spin" /> : <RefreshCw className="h-4 w-4" />}
   149→            </Button>
   150→            <Button variant="outline" size="sm" onClick={onLogout}>
   151→              <LogOut className="h-4 w-4 mr-1.5" /> Sign out
   152→            </Button>
   153→          </div>
   154→        </div>
   155→      </header>
   156→
   157→      {/* Isolation banner */}
   158→      <div className="bg-emerald-50 border-b border-emerald-200">
   159→        <div className="max-w-5xl mx-auto px-4 sm:px-6 py-2 flex items-center gap-2 text-xs text-emerald-800">
   160→          <ShieldCheck className="h-3.5 w-3.5 shrink-0" />
   161→          You are signed in as <strong>{company.name}</strong>. You can only view and edit your own company data.
   162→        </div>
   163→      </div>
   164→
   165→      {/* Tabs */}
   166→      <div className="border-b bg-background">
   167→        <div className="max-w-5xl mx-auto px-4 sm:px-6 flex gap-1 overflow-x-auto">
   168→          <TabBtn active={tab === 'profile'} onClick={() => setTab('profile')} icon={<Building2 className="h-4 w-4" />}>Profile</TabBtn>
   169→          <TabBtn active={tab === 'founders'} onClick={() => setTab('founders')} icon={<Users className="h-4 w-4" />}>Founders {data && <span className="ml-1 text-xs text-muted-foreground">({data.founders.length})</span>}</TabBtn>
   170→          <TabBtn active={tab === 'annual'} onClick={() => setTab('annual')} icon={<DollarSign className="h-4 w-4" />}>Annual Data {data && <span className="ml-1 text-xs text-muted-foreground">({data.annualData.length})</span>}</TabBtn>
   171→        </div>

   174→      <main className="flex-1 max-w-5xl w-full mx-auto px-4 sm:px-6 py-6">

   176→          <div className="py-12 text-center text-muted-foreground text-sm"><Loader2 className="h-5 w-5 animate-spin inline mr-2" />Loading your company data…</div>
   177→        ) : !data ? (
   178→          <div className="py-12 text-center text-muted-foreground text-sm">Failed to load.</div>
   179→        ) : (
   180→          <>
   181→            {tab === 'profile' && <ProfileTab company={data} customColumns={customColumns} onChanged={load} revealed={revealed} setRevealed={setRevealed} />}
   182→            {tab === 'founders' && <FoundersTab company={data} onChanged={load} />}

   190→          SEECS Company Registry · NUST Islamabad
   191→        </div>
   192→      </footer>
   193→    </div>
   194→  )

   285→function ProfileTab({ company, customColumns, onChanged, revealed, setRevealed }: { company: CompanyFull; customColumns: CustomColumn[]; onChanged: () => void; revealed: boolean; setRevealed: (b: boolean) => void }) {
   286→  const [name, setName] = useState(company.name)
   287→  const [description, setDescription] = useState(company.description || '')
   288→  const [email, setEmail] = useState(company.email || '')
   289→  const [phone, setPhone] = useState(company.phone || '')
   290→  const [website, setWebsite] = useState(company.website || '')
   291→  const [locationId, setLocationId] = useState(company.locationId || '')
   292→  const [customValues, setCustomValues] = useState<Record<string, string>>(() => {
   293→    const map: Record<string, string> = {}
   294→    company.customValues?.forEach((cv) => { map[cv.customColumnId] = cv.value })
   295→    return map
   296→  })
   297→  const [lookup, setLookup] = useState<Lookup | null>(null)
   298→  const [saving, setSaving] = useState(false)
   299→  const [savingCustom, setSavingCustom] = useState(false)
   300→  const { toast } = useToast()
   301→
   302→  useEffect(() => {
   303→    api<Lookup>('/api/company/lookup').then(setLookup).catch(() => {})
   304→  }, [])
   305→
   306→  const save = async (e: React.FormEvent) => {
   307→    e.preventDefault()
   308→    setSaving(true)
   309→    try {
   310→      await api('/api/company/self', { method: 'PUT', body: { name, description, email, phone, website, locationId: locationId || null } })
   311→      toast({ title: 'Saved', description: 'Your profile has been updated' })
   312→      onChanged()
   313→    } catch (e: any) {
   314→      toast({ title: 'Error', description: e.message, variant: 'destructive' })
   315→    } finally {
   316→      setSaving(false)
   317→    }
   318→  }
   319→
   320→  const saveCustomValues = async () => {
   321→    setSavingCustom(true)
   322→    try {
   323→      await api('/api/company/self', { method: 'PUT', body: { customValues } })
   324→      toast({ title: 'Saved', description: 'Custom fields updated' })
   325→      onChanged()
   326→    } catch (e: any) {
   327→      toast({ title: 'Error', description: e.message, variant: 'destructive' })
   328→    } finally {
   329→      setSavingCustom(false)
   330→    }
   331→  }
   332→
   333→  return (
   334→    <div className="space-y-5">
   335→      {/* Quick Stats */}
   336→      <QuickStats company={company} />
   337→
   338→      <Card>
   339→        <CardHeader>
   340→          <CardTitle className="text-base">Company profile</CardTitle>
   341→          <CardDescription>Update your company's public information. Sector and city are managed by the SEECS admin.</CardDescription>
   342→        </CardHeader>
   343→        <CardContent>
   344→          <form onSubmit={save} className="space-y-4">
   345→            <div className="grid sm:grid-cols-2 gap-3">
   346→              <div className="space-y-1.5 sm:col-span-2">
   347→                <Label>Company name</Label>
   348→                <Input value={name} onChange={(e) => setName(e.target.value)} required />
   349→              </div>
   350→              <div className="space-y-1.5 sm:col-span-2">
   351→                <Label>Description</Label>
   352→                <Textarea value={description} onChange={(e) => setDescription(e.target.value)} rows={3} />
   353→              </div>
   354→              <div className="space-y-1.5">
   355→                <Label>Email</Label>
   356→                <Input type="email" value={email} onChange={(e) => setEmail(e.target.value)} />
   357→              </div>
   358→              <div className="space-y-1.5">
   359→                <Label>Phone</Label>
   360→                <Input value={phone} onChange={(e) => setPhone(e.target.value)} />
   361→              </div>
   362→              <div className="space-y-1.5 sm:col-span-2">
   363→                <Label>Website</Label>
   364→                <Input value={website} onChange={(e) => setWebsite(e.target.value)} placeholder="https://…" />
   365→              </div>
   366→              <div className="space-y-1.5 sm:col-span-2">
   367→                <Label>Office location</Label>
   368→                <select
   369→                  className="flex h-9 w-full rounded-md border border-input bg-background px-3 py-1 text-sm ring-offset-background focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
   370→                  value={locationId}
   371→                  onChange={(e) => setLocationId(e.target.value)}
   372→                >
   373→                  <option value="">— None —</option>
   374→                  {lookup?.locations.map((l) => <option key={l.id} value={l.id}>{l.address} · {l.city.name}</option>)}
   375→                </select>
   376→              </div>
   377→            </div>
   378→
   379→            <div className="grid sm:grid-cols-2 gap-3 pt-3 border-t">
   380→              <div>
   381→                <Label className="text-xs text-muted-foreground">Sector (managed by admin)</Label>

   385→                <Label className="text-xs text-muted-foreground">City (managed by admin)</Label>
   386→                <div className="text-sm font-medium mt-1">{company.city.name}</div>
   387→              </div>
   388→            </div>
   389→
   390→            {(company.status || company.foundedYear || company.sinceDate || company.branchesCount || company.revenue != null || company.revenueMin != null) && (
   391→              <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-3 pt-3 border-t">
   392→                {company.status && (
   393→                  <div>
   394→                    <Label className="text-xs text-muted-foreground">Status</Label>
   395→                    <div className="mt-1">
   396→                      <Badge variant="outline" className={`text-xs ${STATUS_STYLES[company.status] || ''}`}>{company.status}</Badge>
   397→                    </div>
   398→                  </div>
   399→                )}
   400→                {company.foundedYear && (
   401→                  <div>
   402→                    <Label className="text-xs text-muted-foreground">Founded year</Label>
   403→                    <div className="text-sm font-medium mt-1">{company.foundedYear}</div>
   404→                  </div>
   405→                )}
   406→                {company.sinceDate && (
   407→                  <div>
   408→                    <Label className="text-xs text-muted-foreground">Since date</Label>
   409→                    <div className="text-sm font-medium mt-1">{company.sinceDate}</div>
   410→                  </div>
   411→                )}
   412→                {company.branchesCount != null && (
   413→                  <div>
   414→                    <Label className="text-xs text-muted-foreground">Branches</Label>
   415→                    <div className="text-sm font-medium mt-1">{company.branchesCount}</div>
   416→                  </div>
   417→                )}
   418→                {company.revenue != null ? (
   419→                  <div>
   420→                    <Label className="text-xs text-muted-foreground">Revenue</Label>
   421→                    <div className="text-sm font-medium mt-1">{formatCurrency(company.revenue)}</div>
   422→                  </div>
   423→                ) : (company.revenueMin != null || company.revenueMax != null) ? (

   425→                    <Label className="text-xs text-muted-foreground">Revenue range</Label>
   426→                    <div className="text-sm font-medium mt-1">{formatCurrency(company.revenueMin || 0)} – {formatCurrency(company.revenueMax || 0)}</div>

   429→                {company.statusReason && (
   430→                  <div className="sm:col-span-2">
   431→                    <Label className="text-xs text-muted-foreground">Status reason</Label>

   434→                )}
   435→              </div>

   437→
   438→            <div className="flex justify-end">
   439→              <Button type="submit" disabled={saving}>
   440→                {saving && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   441→                Save changes
   442→              </Button>
   443→            </div>
   444→          </form>
   445→        </CardContent>
   446→      </Card>
   447→
   448→      {/* Custom columns card — dashed border with info icon */}
   449→      {customColumns.length > 0 && (
   450→        <div className="rounded-xl border-2 border-dashed border-muted-foreground/25 bg-muted/20 p-5">
   451→          <div className="flex items-center gap-2 mb-1">
   452→            <Columns3 className="h-4 w-4 text-muted-foreground" />
   453→            <h3 className="text-base font-semibold">Custom fields</h3>
   454→            <div className="ml-auto">
   455→              <Info className="h-4 w-4 text-muted-foreground" />
   456→            </div>
   457→          </div>
   458→          <p className="text-xs text-muted-foreground mb-4">
   459→            Additional fields defined by the SEECS admin. Fill these in to provide extra information about your company.

   461→          <div className="grid sm:grid-cols-2 gap-3">
   462→            {customColumns.map((col) => (
   463→              <div key={col.id} className="space-y-1.5">
   464→                <Label>{col.name}{col.required && <span className="text-destructive ml-1">*</span>}{col.description && <span className="text-xs text-muted-foreground ml-1.5">— {col.description}</span>}</Label>
   465→                {col.columnType === 'number' ? (
   466→                  <Input

   890→function AnnualFormDialog({ open, onOpenChange, existing, onSaved }: { open: boolean; onOpenChange: (o: boolean) => void; existing?: { id: string; year: number; monthlyRevenue: number; totalRevenue: number; employeeCount: number; projectCount: number; notes: string }; onSaved: () => void }) {

   894→  const [employeeCount, setEmployeeCount] = useState('')
