   605→function AnnualFormDialog({ open, onOpenChange, existing, onSaved }: { open: boolean; onOpenChange: (o: boolean) => void; existing?: { id: string; year: number; monthlyRevenue: number; totalRevenue: number; employeeCount: number; projectCount: number; notes: string }; onSaved: () => void }) {
   606→  const [year, setYear] = useState('')
   607→  const [monthlyRevenue, setMonthlyRevenue] = useState('')
   608→  const [totalRevenue, setTotalRevenue] = useState('')
   609→  const [employeeCount, setEmployeeCount] = useState('')
   610→  const [projectCount, setProjectCount] = useState('')
   611→  const [notes, setNotes] = useState('')
   612→  const [saving, setSaving] = useState(false)
   613→  const { toast } = useToast()
   614→
   615→  useEffect(() => {
   616→    if (open) {
   617→      setYear(existing ? String(existing.year) : String(new Date().getFullYear() - 1))
   618→      setMonthlyRevenue(existing ? String(existing.monthlyRevenue) : '')
   619→      setTotalRevenue(existing ? String(existing.totalRevenue) : '')
   620→      setEmployeeCount(existing ? String(existing.employeeCount) : '')
   621→      setProjectCount(existing ? String(existing.projectCount) : '')
   622→      setNotes(existing?.notes || '')
   623→    }
   624→  }, [open, existing])
   625→
   626→  const submit = async (e: React.FormEvent) => {
   627→    e.preventDefault()
   628→    setSaving(true)
   629→    try {
   630→      const body = {
   631→        year: Number(year),
   632→        monthlyRevenue: Number(monthlyRevenue) || 0,
   633→        totalRevenue: Number(totalRevenue) || (Number(monthlyRevenue) || 0) * 12,
   634→        employeeCount: Number(employeeCount) || 0,
   635→        projectCount: Number(projectCount) || 0,
   636→        notes: notes || null,
   637→      }
   638→      if (existing) {
   639→        await api(`/api/company/self/annual-data/${existing.id}`, { method: 'PUT', body })
   640→        toast({ title: 'Saved', description: 'Annual data updated' })
   641→      } else {
   642→        await api('/api/company/self/annual-data', { method: 'POST', body })
   643→        toast({ title: 'Added', description: 'Annual data added' })
   644→      }
   645→      onSaved()
   646→    } catch (e: any) {
   647→      toast({ title: 'Error', description: e.message, variant: 'destructive' })
   648→    } finally {
   649→      setSaving(false)
   650→    }
   651→  }
   652→
   653→  return (
   654→    <Dialog open={open} onOpenChange={onOpenChange}>
   655→      <DialogContent>
   656→        <DialogHeader>
   657→          <DialogTitle>{existing ? 'Edit annual data' : 'Add annual data'}</DialogTitle>
   658→          <DialogDescription>Report your company's financial performance for a specific year.</DialogDescription>
   659→        </DialogHeader>
   660→        <form onSubmit={submit} className="space-y-3">
   661→          <div className="grid grid-cols-2 gap-3">
   662→            <div className="space-y-1.5">
   663→              <Label>Year *</Label>
   664→              <Input type="number" value={year} onChange={(e) => setYear(e.target.value)} required min="2000" max="2100" />
   665→            </div>
   666→            <div className="space-y-1.5">
   667→              <Label>Monthly revenue (PKR)</Label>
   668→              <Input type="number" value={monthlyRevenue} onChange={(e) => setMonthlyRevenue(e.target.value)} placeholder="0" />
   669→            </div>
   670→            <div className="space-y-1.5">
   671→              <Label>Total revenue (PKR)</Label>
   672→              <Input type="number" value={totalRevenue} onChange={(e) => setTotalRevenue(e.target.value)} placeholder="auto = monthly × 12" />
   673→            </div>
   674→            <div className="space-y-1.5">
   675→              <Label>Employees</Label>
   676→              <Input type="number" value={employeeCount} onChange={(e) => setEmployeeCount(e.target.value)} placeholder="0" />
   677→            </div>
   678→            <div className="space-y-1.5">
   679→              <Label>Projects</Label>
   680→              <Input type="number" value={projectCount} onChange={(e) => setProjectCount(e.target.value)} placeholder="0" />
   681→            </div>
   682→            <div className="space-y-1.5 col-span-2">
   683→              <Label>Notes</Label>
   684→              <Textarea value={notes} onChange={(e) => setNotes(e.target.value)} rows={2} />
   685→            </div>
   686→          </div>
   687→          <DialogFooter>
   688→            <Button type="button" variant="ghost" onClick={() => onOpenChange(false)}>Cancel</Button>
   689→            <Button type="submit" disabled={saving}>
   690→              {saving && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   691→              {existing ? 'Save changes' : 'Add year'}
   692→            </Button>
   693→          </DialogFooter>
   694→        </form>
   695→      </DialogContent>
   696→    </Dialog>
   697→  )
   698→}
