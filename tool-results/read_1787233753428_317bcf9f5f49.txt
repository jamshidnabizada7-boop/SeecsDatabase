   342→
   343→function CompanyFormDialog({
   344→  open,
   345→  onOpenChange,
   346→  company,
   347→  lookup,
   348→  customColumns,
   349→  onSaved,
   350→}: {
   351→  open: boolean
   352→  onOpenChange: (o: boolean) => void
   353→  company?: Company | null
   354→  lookup: Lookup | null
   355→  customColumns: CustomColumn[]
   356→  onSaved: () => void
   357→}) {
   358→  const [name, setName] = useState('')
   359→  const [description, setDescription] = useState('')
   360→  const [email, setEmail] = useState('')
   361→  const [phone, setPhone] = useState('')
   362→  const [website, setWebsite] = useState('')
   363→  const [sectorId, setSectorId] = useState('')
   364→  const [cityId, setCityId] = useState('')
   365→  const [locationId, setLocationId] = useState('')
   366→  const [founderIds, setFounderIds] = useState<string[]>([])
   367→  const [apiKeyActive, setApiKeyActive] = useState(true)
   368→  const [regenerateKey, setRegenerateKey] = useState(false)
   369→  // New fields
   370→  const [status, setStatus] = useState('')
   371→  const [statusReason, setStatusReason] = useState('')
   372→  const [sinceDate, setSinceDate] = useState('')
   373→  const [foundedYear, setFoundedYear] = useState('')
   374→  const [discontinuedDate, setDiscontinuedDate] = useState('')
   375→  const [branchesCount, setBranchesCount] = useState('')
   376→  const [revenue, setRevenue] = useState('')
   377→  const [revenueMin, setRevenueMin] = useState('')
   378→  const [revenueMax, setRevenueMax] = useState('')
   379→  // Custom column values
   380→  const [customValues, setCustomValues] = useState<Record<string, string>>({})
   381→  const [saving, setSaving] = useState(false)
   382→  const { toast } = useToast()
   383→
   384→  useEffect(() => {
   385→    if (open) {
   386→      setName(company?.name || '')
   387→      setDescription(company?.description || '')
   388→      setEmail(company?.email || '')
   389→      setPhone(company?.phone || '')
   390→      setWebsite(company?.website || '')
   391→      setSectorId(company?.sectorId || '')
   392→      setCityId(company?.cityId || '')
   393→      setLocationId(company?.locationId || '')
   394→      setFounderIds(company?.founders.map((f) => f.founder.id) || [])
   395→      setApiKeyActive(company?.apiKeyActive ?? true)
   396→      setRegenerateKey(false)
   397→      // New fields
   398→      setStatus(company?.status || '')
   399→      setStatusReason(company?.statusReason || '')
   400→      setSinceDate(company?.sinceDate || '')
   401→      setFoundedYear(company?.foundedYear ? String(company.foundedYear) : '')
   402→      setDiscontinuedDate(company?.discontinuedDate || '')
   403→      setBranchesCount(company?.branchesCount != null ? String(company.branchesCount) : '')
   404→      setRevenue(company?.revenue != null ? String(company.revenue) : '')
   405→      setRevenueMin(company?.revenueMin != null ? String(company.revenueMin) : '')
   406→      setRevenueMax(company?.revenueMax != null ? String(company.revenueMax) : '')
   407→      // Custom values
   408→      const cvMap: Record<string, string> = {}
   409→      company?.customValues?.forEach((cv) => { cvMap[cv.customColumnId] = cv.value })
   410→      setCustomValues(cvMap)
   411→    }
   412→  }, [open, company])
   413→
   414→  const submit = async (e: React.FormEvent) => {
   415→    e.preventDefault()
   416→    if (!name || !sectorId || !cityId) {
   417→      toast({ title: 'Validation', description: 'Name, sector and city are required', variant: 'destructive' })
   418→      return
   419→    }
   420→    // Validate required custom columns
   421→    for (const col of customColumns) {
   422→      if (col.required && !customValues[col.id]?.trim()) {
   423→        toast({ title: 'Validation', description: `"${col.name}" is required`, variant: 'destructive' })
   424→        return
   425→      }
   426→    }
   427→    setSaving(true)
   428→    try {
   429→      const body: Record<string, unknown> = {
   430→        name, description, email, phone, website,
   431→        sectorId, cityId, locationId: locationId || null,
   432→        founderIds, apiKeyActive, regenerateKey,
   433→        // New fields
   434→        status: status || null,
   435→        statusReason: statusReason || null,
   436→        sinceDate: sinceDate || null,
   437→        foundedYear: foundedYear ? Number(foundedYear) : null,
   438→        discontinuedDate: discontinuedDate || null,
   439→        branchesCount: branchesCount ? Number(branchesCount) : null,
   440→        revenue: revenue ? Number(revenue) : null,
   441→        revenueMin: revenueMin ? Number(revenueMin) : null,
   442→        revenueMax: revenueMax ? Number(revenueMax) : null,
   443→      }
   444→      // Add custom values as a separate key
   445→      const cvPayload = customColumns.map((col) => ({
   446→        customColumnId: col.id,
   447→        value: customValues[col.id] || '',
   448→      }))
   449→      if (cvPayload.length > 0) {
   450→        body.customValues = cvPayload
   451→      }
   452→
   453→      if (company) {
   454→        await api(`/api/admin/companies/${company.id}`, { method: 'PUT', body })
   455→        toast({ title: 'Saved', description: 'Company updated' })
   456→      } else {
   457→        await api('/api/admin/companies', { method: 'POST', body })
   458→        toast({ title: 'Created', description: 'Company added' })
   459→      }
   460→      onSaved()
   461→    } catch (e: any) {
   462→      toast({ title: 'Error', description: e.message, variant: 'destructive' })
   463→    } finally {
   464→      setSaving(false)
   465→    }
   466→  }
   467→
   468→  return (
   469→    <Dialog open={open} onOpenChange={onOpenChange}>
   470→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   471→        <DialogHeader>
   472→          <DialogTitle>{company ? 'Edit company' : 'New company'}</DialogTitle>
   473→          <DialogDescription>
   474→            {company ? 'Update the company profile and manage its API key.' : 'Add a new company to the registry. A unique API key is generated automatically.'}
   475→          </DialogDescription>
   476→        </DialogHeader>
   477→        <form onSubmit={submit} className="space-y-4">
   478→          {/* Basic info */}
   479→          <div className="grid sm:grid-cols-2 gap-3">
   480→            <div className="space-y-1.5 sm:col-span-2">
   481→              <Label>Company name *</Label>
   482→              <Input value={name} onChange={(e) => setName(e.target.value)} required />
   483→            </div>
   484→            <div className="space-y-1.5 sm:col-span-2">
   485→              <Label>Description</Label>
   486→              <Textarea value={description} onChange={(e) => setDescription(e.target.value)} rows={2} />
   487→            </div>
   488→            <div className="space-y-1.5">
   489→              <Label>Email</Label>
   490→              <Input type="email" value={email} onChange={(e) => setEmail(e.target.value)} />
   491→            </div>
   492→            <div className="space-y-1.5">
   493→              <Label>Phone</Label>
   494→              <Input value={phone} onChange={(e) => setPhone(e.target.value)} />
   495→            </div>
