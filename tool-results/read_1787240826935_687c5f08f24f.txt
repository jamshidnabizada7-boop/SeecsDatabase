    13→import { Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle } from '@/components/ui/dialog'

    37→interface Company {
    38→  id: string
    39→  name: string
    40→  description: string | null
    41→  email: string | null
    42→  phone: string | null
    43→  website: string | null
    44→  apiKey: string
    45→  apiKeyActive: boolean
    46→  sectorId: string
    47→  cityId: string
    48→  locationId: string | null
    49→  registeredAt: string
    50→  // New Java fields
    51→  status: string | null
    52→  statusReason: string | null
    53→  sinceDate: string | null
    54→  foundedYear: number | null
    55→  discontinuedDate: string | null
    56→  branchesCount: number | null
    57→  revenue: number | null
    58→  revenueMin: number | null
    59→  revenueMax: number | null
    60→  // Relations
    61→  sector: { id: string; name: string }
    62→  city: { id: string; name: string }
    63→  location: { id: string; address: string; country: string; city: { name: string } } | null

    72→  locations: { id: string; address: string; city: { name: string } }[]
    73→  degrees: { id: string; name: string; field: string | null }[]
    74→  founders: { id: string; firstName: string; lastName: string; gender: string }[]

    78→  Active: 'bg-emerald-50 text-emerald-700 border-emerald-200',
    79→  Inactive: 'bg-amber-50 text-amber-700 border-amber-200',
    80→  Discontinued: 'bg-rose-50 text-rose-700 border-rose-200',
    81→}
    82→
    83→type ViewMode = 'card' | 'table'
    84→type StatusFilter = 'All' | 'Active' | 'Inactive' | 'Discontinued'

    86→export default function CompaniesPage() {
    87→  const [items, setItems] = useState<Company[]>([])
    88→  const [customColumns, setCustomColumns] = useState<CustomColumn[]>([])
    89→  const [lookup, setLookup] = useState<Lookup | null>(null)
    90→  const [loading, setLoading] = useState(true)
    91→  const [search, setSearch] = useState('')
    92→  const [editing, setEditing] = useState<Company | null>(null)
    93→  const [creating, setCreating] = useState(false)
    94→  const [deleting, setDeleting] = useState<Company | null>(null)
    95→  const [revealedKeys, setRevealedKeys] = useState<Record<string, boolean>>({})
    96→  const [showCustom, setShowCustom] = useState<Record<string, boolean>>({})
    97→  const [viewMode, setViewMode] = useState<ViewMode>('card')
    98→  const [statusFilter, setStatusFilter] = useState<StatusFilter>('All')
    99→  const { toast } = useToast()
   100→
   101→  const load = async () => {
   102→    setLoading(true)
   103→    try {
   104→      const [companiesRes, lookupRes] = await Promise.all([
   105→        api<{ items: Company[]; customColumns: CustomColumn[] }>(`/api/admin/companies?q=${encodeURIComponent(search)}`),
   106→        api<Lookup>('/api/lookup'),
   107→      ])
   108→      setItems(companiesRes.items)
   109→      setCustomColumns(companiesRes.customColumns)
   110→      setLookup(lookupRes)
   111→    } catch (e: any) {
   112→      toast({ title: 'Error', description: e.message, variant: 'destructive' })
   113→    } finally {
   114→      setLoading(false)
   115→    }
   116→  }
   117→
   118→  useEffect(() => {
   119→    const t = setTimeout(load, 250)
   120→    return () => clearTimeout(t)
   121→  }, [search])
   122→
   123→  const filteredItems = useMemo(() => {
   124→    if (statusFilter === 'All') return items
   125→    return items.filter((c) => c.status === statusFilter)
   126→  }, [items, statusFilter])
   127→
   128→  const copyKey = (k: string) => {
   129→    navigator.clipboard?.writeText(k)
   130→    toast({ title: 'Copied', description: 'API key copied to clipboard' })
   131→  }
   132→
   133→  const regenerateKey = async (company: Company) => {
   134→    try {
   135→      await api(`/api/admin/companies/${company.id}`, {
   136→        method: 'PUT',
   137→        body: { ...company, regenerateKey: true },
   138→      })
   139→      toast({ title: 'Regenerated', description: 'A new API key has been generated' })
   140→      load()
   141→    } catch (e: any) {
   142→      toast({ title: 'Error', description: e.message, variant: 'destructive' })
   143→    }
   144→  }
   145→
   146→  const exportCSV = () => {
   147→    const headers = ['Name', 'Email', 'Phone', 'Website', 'Status', 'Since Date', 'Founded Year', 'Branches', 'Revenue', 'Sector', 'City', 'API Key']
   148→    const rows = filteredItems.map((c) => [
   149→      c.name,
   150→      c.email || '',
   151→      c.phone || '',
   152→      c.website || '',
   153→      c.status || '',
   154→      c.sinceDate || '',
   155→      c.foundedYear || '',
   156→      c.branchesCount || '',
   157→      c.revenue || '',
   158→      c.sector.name,
   159→      c.city.name,
   160→      c.apiKey,
   161→    ])
   162→    const csvContent = [headers, ...rows]
   163→      .map((row) => row.map((cell) => `"${String(cell).replace(/"/g, '""')}"`).join(','))
   164→      .join('\n')
   165→    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' })
   166→    const url = URL.createObjectURL(blob)
   167→    const a = document.createElement('a')
   168→    a.href = url
   169→    a.download = `companies-export-${new Date().toISOString().slice(0, 10)}.csv`
   170→    a.click()
   171→    URL.revokeObjectURL(url)
   172→    toast({ title: 'Exported', description: `${filteredItems.length} companies exported as CSV` })
   173→  }
   174→
   175→  const getCustomValue = (company: Company, columnId: string) => {
   176→    return company.customValues?.find((cv) => cv.customColumnId === columnId)?.value || ''
   177→  }
   178→
   179→  const statusFilters: { label: string; value: StatusFilter; count: number }[] = [
   180→    { label: 'All', value: 'All', count: items.length },
   181→    { label: 'Active', value: 'Active', count: items.filter((c) => c.status === 'Active').length },
   182→    { label: 'Inactive', value: 'Inactive', count: items.filter((c) => c.status === 'Inactive').length },
   183→    { label: 'Discontinued', value: 'Discontinued', count: items.filter((c) => c.status === 'Discontinued').length },
   184→  ]
   185→
   186→  return (
   187→    <div className="p-4 sm:p-6 space-y-4 max-w-7xl mx-auto">
   188→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   189→        <div>
   190→          <h1 className="text-2xl font-semibold">Companies</h1>
   191→          <p className="text-sm text-muted-foreground">{filteredItems.length} shown · master list with per-company API keys.</p>
   192→        </div>
   193→        <div className="flex gap-2 flex-wrap">
   194→          <Button variant="outline" size="sm" onClick={load} disabled={loading}>
   195→            {loading ? <Loader2 className="h-4 w-4 mr-2 animate-spin" /> : <RefreshCw className="h-4 w-4 mr-2" />}
   196→            Refresh
   197→          </Button>
   198→          <Button variant="outline" size="sm" onClick={exportCSV}>
   199→            <Download className="h-4 w-4 mr-2" /> Export CSV
   200→          </Button>
   201→          <Button size="sm" onClick={() => setCreating(true)}>
   202→            <Plus className="h-4 w-4 mr-2" /> Add company
   203→          </Button>
   204→        </div>
   205→      </div>
   206→
   207→      <div className="flex flex-col sm:flex-row gap-3">
   208→        <div className="relative max-w-sm flex-1">
   209→          <Search className="h-4 w-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   210→          <Input
   211→            placeholder="Search by name…"
   212→            value={search}
   213→            onChange={(e) => setSearch(e.target.value)}
   214→            className="pl-9"
   215→          />
   216→        </div>
   217→        <div className="flex items-center gap-1.5">
   218→          {/* Status filter badges */}
   219→          {statusFilters.map((sf) => (
   220→            <button
   221→              key={sf.value}
   222→              onClick={() => setStatusFilter(sf.value)}
   223→              className={`inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full text-xs font-medium border transition-colors ${
   224→                statusFilter === sf.value
   225→                  ? 'bg-primary text-primary-foreground border-primary'
   226→                  : 'bg-background text-muted-foreground border-border hover:bg-muted hover:text-foreground'
   227→              }`}
   228→            >
   229→              {sf.label}
   230→              <span className={`tabular-nums ${statusFilter === sf.value ? 'text-primary-foreground/80' : 'text-muted-foreground/70'}`}>
   231→                {sf.count}
   232→              </span>
   233→            </button>
   234→          ))}
   235→          <div className="w-px h-6 bg-border mx-1" />
   236→          {/* View mode toggle */}
   237→          <Button
   238→            variant={viewMode === 'card' ? 'default' : 'ghost'}
   239→            size="icon"
   240→            className="h-8 w-8"
   241→            onClick={() => setViewMode('card')}
   242→            title="Card view"
   243→          >
   244→            <LayoutGrid className="h-4 w-4" />
   245→          </Button>
   246→          <Button
   247→            variant={viewMode === 'table' ? 'default' : 'ghost'}
   248→            size="icon"
   249→            className="h-8 w-8"
   250→            onClick={() => setViewMode('table')}
   251→            title="Table view"
   252→          >
   253→            <List className="h-4 w-4" />
   254→          </Button>
   255→        </div>
   256→      </div>
   257→
   258→      {loading && items.length === 0 ? (
   259→        <div className="py-12 text-center text-muted-foreground text-sm"><Loader2 className="h-5 w-5 animate-spin inline mr-2" />Loading…</div>
   260→      ) : viewMode === 'card' ? (
   261→        <div className="grid gap-3">
   262→          {filteredItems.map((c) => {
   263→            const revealed = revealedKeys[c.id]
   264→            const showCust = showCustom[c.id]
   265→            return (
   266→              <Card key={c.id} className="overflow-hidden">
   267→                <CardContent className="p-4 sm:p-5">
   268→                  <div className="flex flex-col lg:flex-row gap-4">
   269→                    {/* Left: identity */}
   270→                    <div className="flex-1 min-w-0">
   271→                      <div className="flex items-start justify-between gap-3">
   272→                        <div className="min-w-0">
   273→                          <div className="flex items-center gap-2 flex-wrap">
   274→                            <h3 className="font-semibold truncate">{c.name}</h3>
   275→                            {c.status ? (
   276→                              <Badge variant="outline" className={`text-xs ${STATUS_STYLES[c.status] || ''}`}>{c.status}</Badge>
   277→                            ) : c.apiKeyActive ? (
   278→                              <Badge variant="secondary" className="text-emerald-600 bg-emerald-50 border-emerald-200">Active</Badge>
   279→                            ) : (
   280→                              <Badge variant="destructive" className="bg-rose-50 text-rose-700 border-rose-200">Revoked</Badge>
   281→                            )}
   282→                          </div>
   283→                          <p className="text-sm text-muted-foreground mt-1 line-clamp-2">{c.description || 'No description.'}</p>
   284→                        </div>
   285→                      </div>
   286→
   287→                      <div className="mt-3 grid grid-cols-2 sm:grid-cols-4 gap-3 text-sm">
   288→                        <Field label="Sector" value={c.sector.name} />
   289→                        <Field label="City" value={c.city.name} />
   290→                        <Field label="Founders" value={String(c.founders.length)} />
   291→                        <Field label="Annual records" value={String(c._count.annualData)} />
   292→                      </div>
   293→
   294→                      {/* New fields row */}
   295→                      {(c.foundedYear || c.sinceDate || c.branchesCount || c.revenue != null || c.revenueMin != null) && (
   296→                        <div className="mt-3 grid grid-cols-2 sm:grid-cols-4 gap-3 text-sm">
   297→                          {c.foundedYear && <Field label="Founded" value={String(c.foundedYear)} />}
   298→                          {c.sinceDate && <Field label="Since" value={c.sinceDate} />}
   299→                          {c.branchesCount != null && <Field label="Branches" value={String(c.branchesCount)} />}
   300→                          {c.revenue != null ? (
   301→                            <Field label="Revenue" value={formatCurrency(c.revenue)} />
   302→                          ) : c.revenueMin != null || c.revenueMax != null ? (
   303→                            <Field label="Revenue range" value={`${formatCurrency(c.revenueMin || 0)} – ${formatCurrency(c.revenueMax || 0)}`} />
   304→                          ) : null}
   305→                        </div>
   306→                      )}
   307→
   308→                      {c.statusReason && (
   309→                        <p className="mt-2 text-xs text-muted-foreground">Status reason: {c.statusReason}</p>
   310→                      )}
   311→                      {c.discontinuedDate && (
   312→                        <p className="mt-1 text-xs text-rose-600">Discontinued: {c.discontinuedDate}</p>
   313→                      )}
   314→
   315→                      {c.founders.length > 0 && (
   316→                        <div className="mt-3 flex flex-wrap gap-1.5">
   317→                          {c.founders.map((cf) => (
   318→                            <Badge key={cf.id} variant="outline" className="text-xs">
   319→                              {cf.founder.firstName} {cf.founder.lastName} · <span className="text-muted-foreground">{cf.role}</span>
   320→                            </Badge>
   321→                          ))}
   322→                        </div>
   323→                      )}
   324→
   325→                      {/* Custom columns inline */}
   326→                      {customColumns.length > 0 && (
   327→                        <div className="mt-3">
   328→                          <button
   329→                            className="flex items-center gap-1.5 text-xs font-medium text-muted-foreground hover:text-foreground transition-colors"
   330→                            onClick={() => setShowCustom((s) => ({ ...s, [c.id]: !s[c.id] }))}
   331→                          >
   332→                            <Columns3 className="h-3.5 w-3.5" />
   333→                            Custom fields ({customColumns.length})
   334→                            {showCust ? <ChevronUp className="h-3 w-3" /> : <ChevronDown className="h-3 w-3" />}
   335→                          </button>
   336→                          {showCust && (
   337→                            <div className="mt-2 grid grid-cols-2 sm:grid-cols-3 gap-2">
   338→                              {customColumns.map((col) => (
   339→                                <div key={col.id} className="text-sm">
   340→                                  <span className="text-xs text-muted-foreground">{col.name}</span>
   341→                                  <div className="font-medium truncate">{getCustomValue(c, col.id) || <span className="text-muted-foreground italic">—</span>}</div>
   342→                                </div>
   343→                              ))}
   344→                            </div>
   345→                          )}
   346→                        </div>
   347→                      )}
   348→                    </div>
   349→
   350→                    {/* Right: API key + actions */}
   351→                    <div className="lg:w-80 lg:border-l lg:pl-5 space-y-2">
   352→                      <div>
   353→                        <Label className="text-xs text-muted-foreground">API Key</Label>
   354→                        <div className="flex items-center gap-1 mt-1">
   355→                          <code className="text-xs font-mono bg-muted px-2 py-1 rounded flex-1 truncate">
   356→                            {revealed ? c.apiKey : `${c.apiKey.slice(0, 14)}••••••••`}
   357→                          </code>
   358→                          <Tooltip>
   359→                            <TooltipTrigger asChild>
   360→                              <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => setRevealedKeys((s) => ({ ...s, [c.id]: !s[c.id] }))}>
   361→                                {revealed ? <EyeOff className="h-3.5 w-3.5" /> : <Eye className="h-3.5 w-3.5" />}
   362→                              </Button>
   363→                            </TooltipTrigger>
   364→                            <TooltipContent>{revealed ? 'Hide key' : 'Reveal key'}</TooltipContent>
   365→                          </Tooltip>
   366→                          <Tooltip>
   367→                            <TooltipTrigger asChild>
   368→                              <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => copyKey(c.apiKey)}>
   369→                                <Copy className="h-3.5 w-3.5" />
   370→                              </Button>
   371→                            </TooltipTrigger>
   372→                            <TooltipContent>Copy to clipboard</TooltipContent>
   373→                          </Tooltip>
   374→                          <Tooltip>
   375→                            <TooltipTrigger asChild>
   376→                              <Button variant="ghost" size="icon" className="h-7 w-7 text-amber-600 hover:text-amber-700 hover:bg-amber-50" onClick={() => regenerateKey(c)}>
   377→                                <RotateCw className="h-3.5 w-3.5" />
   378→                              </Button>
   379→                            </TooltipTrigger>
   380→                            <TooltipContent>Regenerate key</TooltipContent>
   381→                          </Tooltip>
   382→                        </div>
   383→                      </div>
   384→                      <div className="flex gap-2">
   385→                        <Button variant="outline" size="sm" className="flex-1" onClick={() => setEditing(c)}>
   386→                          <Pencil className="h-3.5 w-3.5 mr-1" /> Edit
   387→                        </Button>
   388→                        <Button variant="outline" size="sm" className="text-destructive hover:bg-destructive/5" onClick={() => setDeleting(c)}>
   389→                          <Trash2 className="h-3.5 w-3.5" />
   390→                        </Button>
   391→                      </div>
   392→                    </div>
   393→                  </div>
   394→                </CardContent>
   395→              </Card>
   396→            )
   397→          })}
   398→          {filteredItems.length === 0 && (
   399→            <div className="py-16 text-center text-muted-foreground">
   400→              <Building2 className="h-8 w-8 mx-auto mb-3 opacity-50" />
   401→              {statusFilter !== 'All' ? `No companies with status "${statusFilter}".` : 'No companies yet. Add one or let companies self-register.'}
   402→            </div>
   403→          )}
   404→        </div>
   405→      ) : (
   406→        /* Table View */
   407→        <Card>
   408→          <CardContent className="p-0">
   409→            <Table>
   410→              <TableHeader>
   411→                <TableRow>
   412→                  <TableHead className="pl-4">Name</TableHead>
   413→                  <TableHead>Sector</TableHead>
   414→                  <TableHead className="hidden md:table-cell">City</TableHead>
   415→                  <TableHead>Status</TableHead>
   416→                  <TableHead className="hidden lg:table-cell">Founded</TableHead>
   417→                  <TableHead className="hidden lg:table-cell">Revenue</TableHead>
   418→                  <TableHead>API Key</TableHead>
   419→                  <TableHead className="text-right pr-4">Actions</TableHead>
   420→                </TableRow>
   421→              </TableHeader>
   422→              <TableBody>
   423→                {filteredItems.map((c) => {
   424→                  const revealed = revealedKeys[c.id]
   425→                  return (
   426→                    <TableRow key={c.id}>
   427→                      <TableCell className="pl-4 font-medium max-w-[200px] truncate">
   428→                        <div className="truncate">{c.name}</div>
   429→                        {c.email && <div className="text-xs text-muted-foreground truncate">{c.email}</div>}
   430→                      </TableCell>
   431→                      <TableCell className="text-muted-foreground">{c.sector.name}</TableCell>
   432→                      <TableCell className="hidden md:table-cell text-muted-foreground">{c.city.name}</TableCell>
   433→                      <TableCell>
   434→                        {c.status ? (
   435→                          <Badge variant="outline" className={`text-xs ${STATUS_STYLES[c.status] || ''}`}>{c.status}</Badge>
   436→                        ) : c.apiKeyActive ? (
   437→                          <Badge variant="secondary" className="text-xs text-emerald-600 bg-emerald-50 border-emerald-200">Active</Badge>
   438→                        ) : (
   439→                          <Badge variant="destructive" className="text-xs bg-rose-50 text-rose-700 border-rose-200">Revoked</Badge>
   440→                        )}
   441→                      </TableCell>
   442→                      <TableCell className="hidden lg:table-cell text-muted-foreground">
   443→                        {c.foundedYear || '—'}
   444→                      </TableCell>
   445→                      <TableCell className="hidden lg:table-cell text-muted-foreground">
   446→                        {c.revenue != null ? formatCurrency(c.revenue) : '—'}
   447→                      </TableCell>
   448→                      <TableCell>
   449→                        <div className="flex items-center gap-0.5">
   450→                          <code className="text-xs font-mono bg-muted px-1.5 py-0.5 rounded max-w-[120px] truncate">
   451→                            {revealed ? c.apiKey : `${c.apiKey.slice(0, 10)}•••`}
   452→                          </code>
   453→                          <Tooltip>
   454→                            <TooltipTrigger asChild>
   455→                              <Button variant="ghost" size="icon" className="h-6 w-6" onClick={() => setRevealedKeys((s) => ({ ...s, [c.id]: !s[c.id] }))}>
   456→                                {revealed ? <EyeOff className="h-3 w-3" /> : <Eye className="h-3 w-3" />}
   457→                              </Button>
   458→                            </TooltipTrigger>
   459→                            <TooltipContent>{revealed ? 'Hide' : 'Reveal'}</TooltipContent>
   460→                          </Tooltip>
   461→                          <Tooltip>
   462→                            <TooltipTrigger asChild>
   463→                              <Button variant="ghost" size="icon" className="h-6 w-6" onClick={() => copyKey(c.apiKey)}>
   464→                                <Copy className="h-3 w-3" />
   465→                              </Button>
   466→                            </TooltipTrigger>
   467→                            <TooltipContent>Copy</TooltipContent>
   468→                          </Tooltip>
   469→                          <Tooltip>
   470→                            <TooltipTrigger asChild>
   471→                              <Button variant="ghost" size="icon" className="h-6 w-6 text-amber-600 hover:text-amber-700" onClick={() => regenerateKey(c)}>
   472→                                <RotateCw className="h-3 w-3" />
   473→                              </Button>
   474→                            </TooltipTrigger>
   475→                            <TooltipContent>Regenerate</TooltipContent>
   476→                          </Tooltip>
   477→                        </div>
   478→                      </TableCell>
   479→                      <TableCell className="text-right pr-4">
   480→                        <div className="flex items-center justify-end gap-1">
   481→                          <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => setEditing(c)}>
   482→                            <Pencil className="h-3.5 w-3.5" />
   483→                          </Button>
   484→                          <Button variant="ghost" size="icon" className="h-7 w-7 text-destructive hover:text-destructive" onClick={() => setDeleting(c)}>
   485→                            <Trash2 className="h-3.5 w-3.5" />
   486→                          </Button>
   487→                        </div>
   488→                      </TableCell>
   489→                    </TableRow>
   490→                  )
   491→                })}
   492→              </TableBody>
   493→            </Table>
   494→            {filteredItems.length === 0 && (
   495→              <div className="py-16 text-center text-muted-foreground">
   496→                <Building2 className="h-8 w-8 mx-auto mb-3 opacity-50" />
   497→                {statusFilter !== 'All' ? `No companies with status "${statusFilter}".` : 'No companies yet.'}
   498→              </div>
   499→            )}
   500→          </CardContent>
   501→        </Card>
   502→      )}
   503→
   504→      {/* Create dialog */}
   505→      <CompanyFormDialog
   506→        open={creating}
   507→        onOpenChange={setCreating}
   508→        lookup={lookup}
   509→        customColumns={customColumns}
   510→        onSaved={() => { setCreating(false); load() }}
   511→      />
   512→      {/* Edit dialog */}
   513→      <CompanyFormDialog
   514→        open={!!editing}
   515→        onOpenChange={(o) => !o && setEditing(null)}
   516→        company={editing}
   517→        lookup={lookup}
   518→        customColumns={customColumns}
   519→        onSaved={() => { setEditing(null); load() }}
   520→      />
   521→      {/* Delete confirm */}
   522→      <AlertDialog open={!!deleting} onOpenChange={(o) => !o && setDeleting(null)}>
   523→        <AlertDialogContent>
   524→          <AlertDialogHeader>
   525→            <AlertDialogTitle>Delete company "{deleting?.name}"?</AlertDialogTitle>
   526→            <AlertDialogDescription>
   527→              This permanently removes the company, its founder attachments and all annual data. This cannot be undone.
   528→            </AlertDialogDescription>
   529→          </AlertDialogHeader>
   530→          <AlertDialogFooter>
   531→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   532→            <AlertDialogAction
   533→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   534→              onClick={async () => {
   535→                try {
   536→                  await api(`/api/admin/companies/${deleting!.id}`, { method: 'DELETE' })
   537→                  toast({ title: 'Deleted', description: `${deleting!.name} removed` })
   538→                  setDeleting(null)
   539→                  load()
   540→                } catch (e: any) {
   541→                  toast({ title: 'Error', description: e.message, variant: 'destructive' })
   542→                }
   543→              }}
   544→            >
   545→              Delete
   546→            </AlertDialogAction>
   547→          </AlertDialogFooter>
   548→        </AlertDialogContent>
   549→      </AlertDialog>
   550→    </div>
   551→  )
   552→}
