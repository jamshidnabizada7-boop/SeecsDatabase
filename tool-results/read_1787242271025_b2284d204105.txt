    46→interface Company {

    48→  name: string
    49→  description: string | null
    50→  email: string | null
    51→  phone: string | null
    52→  website: string | null
    53→  apiKey: string

    55→  sectorId: string
    56→  cityId: string
    57→  locationId: string | null
    58→  registeredAt: string
    59→  // New Java fields
    60→  status: string | null

    62→  sinceDate: string | null
    63→  foundedYear: number | null
    64→  discontinuedDate: string | null

    84→}

    88→  Inactive: 'bg-amber-50 dark:bg-amber-950 text-amber-700 dark:text-amber-300 border-amber-200 dark:border-amber-800',

    90→}

    95→export default function CompaniesPage() {
    96→  const [items, setItems] = useState<Company[]>([])
    97→  const [customColumns, setCustomColumns] = useState<CustomColumn[]>([])
    98→  const [lookup, setLookup] = useState<Lookup | null>(null)
    99→  const [loading, setLoading] = useState(true)
   100→  const [search, setSearch] = useState('')
   101→  const [editing, setEditing] = useState<Company | null>(null)
   102→  const [creating, setCreating] = useState(false)
   103→  const [deleting, setDeleting] = useState<Company | null>(null)
   104→  const [revealedKeys, setRevealedKeys] = useState<Record<string, boolean>>({})
   105→  const [showCustom, setShowCustom] = useState<Record<string, boolean>>({})
   106→  const [viewMode, setViewMode] = useState<ViewMode>('card')
   107→  const [statusFilter, setStatusFilter] = useState<StatusFilter>('All')
   108→  const [page, setPage] = useState(1)
   109→  const [pageSize, setPageSize] = useState(20)
   110→  const [total, setTotal] = useState(0)
   111→  const [totalPages, setTotalPages] = useState(1)
   112→  const { toast } = useToast()
   113→  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   114→
   115→  const buildParams = useCallback((
   116→    opts?: { searchVal?: string; statusVal?: StatusFilter; pageVal?: number; pageSizeVal?: number }
   117→  ) => {
   118→    const q = opts?.searchVal ?? search
   119→    const st = opts?.statusVal ?? statusFilter
   120→    const p = opts?.pageVal ?? page
   121→    const ps = opts?.pageSizeVal ?? pageSize
   122→    const params = new URLSearchParams()
   123→    if (p) params.set('page', String(p))
   124→    if (ps) params.set('limit', String(ps))
   125→    if (q) params.set('q', q)
   126→    if (st !== 'All') params.set('status', st)
   127→    return params.toString() ? `?${params.toString()}` : ''
   128→  }, [search, statusFilter, page, pageSize])
   129→
   130→  const load = useCallback(async (overrideParams?: string) => {
   131→    setLoading(true)
   132→    try {
   133→      const qs = overrideParams ?? buildParams()
   134→      const [companiesRes, lookupRes] = await Promise.all([
   135→        api<{ items: Company[]; customColumns: CustomColumn[]; total: number; page: number; limit: number; totalPages: number }>(`/api/admin/companies${qs}`),
   136→        api<Lookup>('/api/lookup'),
   137→      ])
   138→      setItems(companiesRes.items)
   139→      setCustomColumns(companiesRes.customColumns)
   140→      setLookup(lookupRes)
   141→      setTotal(companiesRes.total)
   142→      setTotalPages(companiesRes.totalPages)
   143→    } catch (e: any) {
   144→      toast({ title: 'Error', description: e.message, variant: 'destructive' })
   145→    } finally {
   146→      setLoading(false)
   147→    }
   148→  }, [buildParams, toast])
   149→
   150→  // Main data loading: debounced for search, immediate for page/status/pageSize changes
   151→  useEffect(() => {
   152→    if (debounceRef.current) clearTimeout(debounceRef.current)
   153→    const delay = search ? 300 : 0
   154→    debounceRef.current = setTimeout(() => {
   155→      load()
   156→    }, delay)
   157→    return () => { if (debounceRef.current) clearTimeout(debounceRef.current) }
   158→  // eslint-disable-next-line react-hooks/exhaustive-deps
   159→  }, [search, statusFilter, page, pageSize])
   160→
   161→  const handlePageSizeChange = (val: string) => {
   162→    setPageSize(Number(val))
   163→    setPage(1)
   164→  }
   165→
   166→  const handleStatusFilter = (val: StatusFilter) => {
   167→    setStatusFilter(val)
   168→    setPage(1)
   169→  }
   170→
   171→  const handleSearchClear = () => {
   172→    setSearch('')
   173→    setPage(1)
   174→  }
   175→
   176→  const copyKey = (k: string) => {
   177→    navigator.clipboard?.writeText(k)
   178→    toast({ title: 'Copied', description: 'API key copied to clipboard' })
   179→  }
   180→
   181→  const regenerateKey = async (company: Company) => {
   182→    try {
   183→      await api(`/api/admin/companies/${company.id}`, {
   184→        method: 'PUT',
   185→        body: { ...company, regenerateKey: true },
   186→      })
   187→      toast({ title: 'Regenerated', description: 'A new API key has been generated' })
   188→      load()
   189→    } catch (e: any) {
   190→      toast({ title: 'Error', description: e.message, variant: 'destructive' })
   191→    }
   192→  }
   193→
   194→  const exportCSV = async () => {
   195→    let exportItems: Company[] = items
   196→    try {
   197→      const exportParams = new URLSearchParams()
   198→      if (search) exportParams.set('q', search)
   199→      if (statusFilter !== 'All') exportParams.set('status', statusFilter)
   200→      const qs = exportParams.toString() ? `?${exportParams.toString()}` : ''
   201→      const res = await api<{ items: Company[] }>(`/api/admin/companies${qs}`)
   202→      exportItems = res.items
   203→    } catch {
   204→      // fall back to current page items
   205→    }
   206→    const headers = ['Name', 'Email', 'Phone', 'Website', 'Status', 'Since Date', 'Founded Year', 'Branches', 'Revenue', 'Sector', 'City', 'API Key']
   207→    const rows = exportItems.map((c) => [
   208→      c.name,
   209→      c.email || '',
   210→      c.phone || '',
   211→      c.website || '',
   212→      c.status || '',
   213→      c.sinceDate || '',
   214→      c.foundedYear || '',
   215→      c.branchesCount || '',
   216→      c.revenue || '',
   217→      c.sector.name,
   218→      c.city.name,
   219→      c.apiKey,
   220→    ])
   221→    const csvContent = [headers, ...rows]
   222→      .map((row) => row.map((cell) => `"${String(cell).replace(/"/g, '""')}"`).join(','))
   223→      .join('\n')
   224→    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' })
   225→    const url = URL.createObjectURL(blob)
   226→    const a = document.createElement('a')
   227→    a.href = url
   228→    a.download = `companies-export-${new Date().toISOString().slice(0, 10)}.csv`
   229→    a.click()
   230→    URL.revokeObjectURL(url)
   231→    toast({ title: 'Exported', description: `${exportItems.length} companies exported as CSV` })
   232→  }
   233→
   234→  const getCustomValue = (company: Company, columnId: string) => {
   235→    return company.customValues?.find((cv) => cv.customColumnId === columnId)?.value || ''
   236→  }
   237→
   238→  const statusFilterOptions: { label: string; value: StatusFilter }[] = [
   239→    { label: 'All', value: 'All' },
   240→    { label: 'Active', value: 'Active' },
   241→    { label: 'Inactive', value: 'Inactive' },
   242→    { label: 'Discontinued', value: 'Discontinued' },
   243→  ]
   244→
   245→  const showingFrom = total === 0 ? 0 : (page - 1) * pageSize + 1
   246→  const showingTo = Math.min(page * pageSize, total)
   247→
   248→  const paginationRange = useMemo(() => {
   249→    const range: number[] = []
   250→    const delta = 2
   251→    const left = Math.max(1, page - delta)
   252→    const right = Math.min(totalPages, page + delta)
   253→    for (let i = left; i <= right; i++) range.push(i)
   254→    return range
   255→  }, [page, totalPages])
   256→
   257→  return (
   258→    <div className="p-4 sm:p-6 space-y-4 max-w-7xl mx-auto">
   259→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   260→        <div>
   261→          <h1 className="text-2xl font-semibold">Companies</h1>
   262→          <p className="text-sm text-muted-foreground">{total} companies total · {statusFilter !== 'All' ? statusFilter : ''} {search ? `matching "${search}"` : ''}</p>
   263→        </div>
   264→        <div className="flex gap-2 flex-wrap">
   265→          <Button variant="outline" size="sm" onClick={load} disabled={loading}>
   266→            {loading ? <Loader2 className="h-4 w-4 mr-2 animate-spin" /> : <RefreshCw className="h-4 w-4 mr-2" />}
   267→            Refresh
   268→          </Button>
   269→          <Button variant="outline" size="sm" onClick={exportCSV}>
   270→            <Download className="h-4 w-4 mr-2" /> Export CSV
   271→          </Button>
   272→          <Button size="sm" onClick={() => setCreating(true)}>
   273→            <Plus className="h-4 w-4 mr-2" /> Add company
   274→          </Button>
   275→        </div>
   276→      </div>
   277→
   278→      <div className="flex flex-col sm:flex-row gap-3">
   279→        <div className="relative max-w-sm flex-1">
   280→          <Search className="h-4 w-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   281→          <Input
   282→            placeholder="Search by name, description, website…"
   283→            value={search}
   284→            onChange={(e) => setSearch(e.target.value)}
   285→            className="pl-9 pr-8"
   286→          />
   287→          {search && (
   288→            <button
   289→              onClick={handleSearchClear}
   290→              className="absolute right-2.5 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground transition-colors"
   291→            >
   292→              <X className="h-3.5 w-3.5" />
   293→            </button>
   294→          )}
   295→        </div>
   296→        <div className="flex items-center gap-1.5 flex-wrap">
   297→          {/* Status filter badges */}
   298→          {statusFilterOptions.map((sf) => (
   299→            <button
   300→              key={sf.value}
   301→              onClick={() => handleStatusFilter(sf.value)}
   302→              className={`inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full text-xs font-medium border transition-colors ${
   303→                statusFilter === sf.value
   304→                  ? 'bg-emerald-600 text-white border-emerald-600'
   305→                  : 'bg-background text-stone-600 border-stone-200 hover:bg-stone-50 hover:text-stone-900'
   306→              }`}
   307→            >
   308→              {sf.label}
   309→            </button>
   310→          ))}
   311→          <div className="w-px h-6 bg-border mx-1" />
   312→          {/* View mode toggle */}
   313→          <Button
   314→            variant={viewMode === 'card' ? 'default' : 'ghost'}
   315→            size="icon"
   316→            className="h-8 w-8"
   317→            onClick={() => setViewMode('card')}
   318→            title="Card view"
   319→          >
   320→            <LayoutGrid className="h-4 w-4" />
   321→          </Button>
   322→          <Button
   323→            variant={viewMode === 'table' ? 'default' : 'ghost'}
   324→            size="icon"
   325→            className="h-8 w-8"
   326→            onClick={() => setViewMode('table')}
   327→            title="Table view"
   328→          >
   329→            <List className="h-4 w-4" />
   330→          </Button>
   331→        </div>
   332→      </div>
   333→
   334→      {loading && items.length === 0 ? (
   335→        <div className="py-12 text-center text-muted-foreground text-sm"><Loader2 className="h-5 w-5 animate-spin inline mr-2" />Loading…</div>
   336→      ) : viewMode === 'card' ? (
   337→        <div className="grid gap-3">
   338→          {items.map((c) => {
   339→            const revealed = revealedKeys[c.id]
   340→            const showCust = showCustom[c.id]
   341→            return (
   342→              <Card key={c.id} className="overflow-hidden">
   343→                <CardContent className="p-4 sm:p-5">
   344→                  <div className="flex flex-col lg:flex-row gap-4">
   345→                    {/* Left: identity */}
   346→                    <div className="flex-1 min-w-0">
   347→                      <div className="flex items-start justify-between gap-3">
   348→                        <div className="min-w-0">
   349→                          <div className="flex items-center gap-2 flex-wrap">
   350→                            <h3 className="font-semibold truncate">{c.name}</h3>
   351→                            {c.status ? (
   352→                              <Badge variant="outline" className={`text-xs ${STATUS_STYLES[c.status] || ''}`}>{c.status}</Badge>
   353→                            ) : c.apiKeyActive ? (
   354→                              <Badge variant="secondary" className="text-emerald-600 bg-emerald-50 border-emerald-200">Active</Badge>
   355→                            ) : (
   356→                              <Badge variant="destructive" className="bg-rose-50 text-rose-700 border-rose-200">Revoked</Badge>
   357→                            )}
   358→                          </div>
   359→                          <p className="text-sm text-muted-foreground mt-1 line-clamp-2">{c.description || 'No description.'}</p>
   360→                        </div>
   361→                      </div>
   362→
   363→                      <div className="mt-3 grid grid-cols-2 sm:grid-cols-4 gap-3 text-sm">
   364→                        <Field label="Sector" value={c.sector.name} />
   365→                        <Field label="City" value={c.city.name} />
   366→                        <Field label="Founders" value={String(c.founders.length)} />
   367→                        <Field label="Annual records" value={String(c._count.annualData)} />
   368→                      </div>
   369→
   370→                      {/* New fields row */}
   371→                      {(c.foundedYear || c.sinceDate || c.branchesCount || c.revenue != null || c.revenueMin != null) && (
   372→                        <div className="mt-3 grid grid-cols-2 sm:grid-cols-4 gap-3 text-sm">
   373→                          {c.foundedYear && <Field label="Founded" value={String(c.foundedYear)} />}
   374→                          {c.sinceDate && <Field label="Since" value={c.sinceDate} />}
   375→                          {c.branchesCount != null && <Field label="Branches" value={String(c.branchesCount)} />}
   376→                          {c.revenue != null ? (
   377→                            <Field label="Revenue" value={formatCurrency(c.revenue)} />
   378→                          ) : c.revenueMin != null || c.revenueMax != null ? (
   379→                            <Field label="Revenue range" value={`${formatCurrency(c.revenueMin || 0)} – ${formatCurrency(c.revenueMax || 0)}`} />
   380→                          ) : null}
   381→                        </div>
   382→                      )}
   383→
   384→                      {c.statusReason && (
   385→                        <p className="mt-2 text-xs text-muted-foreground">Status reason: {c.statusReason}</p>
   386→                      )}
   387→                      {c.discontinuedDate && (
   388→                        <p className="mt-1 text-xs text-rose-600">Discontinued: {c.discontinuedDate}</p>
   389→                      )}
   390→
   391→                      {c.founders.length > 0 && (
   392→                        <div className="mt-3 flex flex-wrap gap-1.5">
   393→                          {c.founders.map((cf) => (
   394→                            <Badge key={cf.id} variant="outline" className="text-xs">
   395→                              {cf.founder.firstName} {cf.founder.lastName} · <span className="text-muted-foreground">{cf.role}</span>
   396→                            </Badge>
   397→                          ))}
   398→                        </div>
   399→                      )}
   400→
   401→                      {/* Custom columns inline */}
   402→                      {customColumns.length > 0 && (
   403→                        <div className="mt-3">
   404→                          <button
   405→                            className="flex items-center gap-1.5 text-xs font-medium text-muted-foreground hover:text-foreground transition-colors"
   406→                            onClick={() => setShowCustom((s) => ({ ...s, [c.id]: !s[c.id] }))}
   407→                          >
   408→                            <Columns3 className="h-3.5 w-3.5" />
   409→                            Custom fields ({customColumns.length})
   410→                            {showCust ? <ChevronUp className="h-3 w-3" /> : <ChevronDown className="h-3 w-3" />}
   411→                          </button>
   412→                          {showCust && (
   413→                            <div className="mt-2 grid grid-cols-2 sm:grid-cols-3 gap-2">
   414→                              {customColumns.map((col) => (
   415→                                <div key={col.id} className="text-sm">
   416→                                  <span className="text-xs text-muted-foreground">{col.name}</span>
   417→                                  <div className="font-medium truncate">{getCustomValue(c, col.id) || <span className="text-muted-foreground italic">—</span>}</div>
   418→                                </div>
   419→                              ))}
   420→                            </div>
   421→                          )}
   422→                        </div>
   423→                      )}
   424→                    </div>
   425→
   426→                    {/* Right: API key + actions */}
   427→                    <div className="lg:w-80 lg:border-l lg:pl-5 space-y-2">
   428→                      <div>
   429→                        <Label className="text-xs text-muted-foreground">API Key</Label>
   430→                        <div className="flex items-center gap-1 mt-1">
   431→                          <code className="text-xs font-mono bg-muted px-2 py-1 rounded flex-1 truncate">
   432→                            {revealed ? c.apiKey : `${c.apiKey.slice(0, 14)}••••••••`}
   433→                          </code>
   434→                          <Tooltip>
   435→                            <TooltipTrigger asChild>
   436→                              <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => setRevealedKeys((s) => ({ ...s, [c.id]: !s[c.id] }))}>
   437→                                {revealed ? <EyeOff className="h-3.5 w-3.5" /> : <Eye className="h-3.5 w-3.5" />}
   438→                              </Button>
   439→                            </TooltipTrigger>
   440→                            <TooltipContent>{revealed ? 'Hide key' : 'Reveal key'}</TooltipContent>
   441→                          </Tooltip>
   442→                          <Tooltip>
   443→                            <TooltipTrigger asChild>
   444→                              <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => copyKey(c.apiKey)}>
   445→                                <Copy className="h-3.5 w-3.5" />
   446→                              </Button>
   447→                            </TooltipTrigger>
   448→                            <TooltipContent>Copy to clipboard</TooltipContent>
   449→                          </Tooltip>
   450→                          <Tooltip>
   451→                            <TooltipTrigger asChild>
   452→                              <Button variant="ghost" size="icon" className="h-7 w-7 text-amber-600 hover:text-amber-700 hover:bg-amber-50" onClick={() => regenerateKey(c)}>
   453→                                <RotateCw className="h-3.5 w-3.5" />
   454→                              </Button>
   455→                            </TooltipTrigger>
   456→                            <TooltipContent>Regenerate key</TooltipContent>
   457→                          </Tooltip>
   458→                        </div>
   459→                      </div>
   460→                      <div className="flex gap-2">
   461→                        <Button variant="outline" size="sm" className="flex-1" onClick={() => setEditing(c)}>
   462→                          <Pencil className="h-3.5 w-3.5 mr-1" /> Edit
   463→                        </Button>
   464→                        <Button variant="outline" size="sm" className="text-destructive hover:bg-destructive/5" onClick={() => setDeleting(c)}>
   465→                          <Trash2 className="h-3.5 w-3.5" />
   466→                        </Button>
   467→                      </div>
   468→                    </div>
   469→                  </div>
   470→                </CardContent>
   471→              </Card>
   472→            )
   473→          })}
   474→          {items.length === 0 && (
   475→            <div className="py-16 text-center text-muted-foreground">
   476→              <Building2 className="h-8 w-8 mx-auto mb-3 opacity-50" />
   477→              {search
   478→                ? 'No companies found matching your search.'
   479→                : statusFilter !== 'All'
   480→                  ? `No companies with status "${statusFilter}".`
   481→                  : 'No companies yet. Add one or let companies self-register.'}

   563→                          </Button>

   568→                      </TableCell>
   569→                    </TableRow>
   570→                  )
   571→                })}
   572→              </TableBody>
   573→            </Table>
   574→            {items.length === 0 && (
   575→              <div className="py-16 text-center text-muted-foreground">
   576→                <Building2 className="h-8 w-8 mx-auto mb-3 opacity-50" />
   577→                {search
   578→                  ? 'No companies found matching your search.'
   579→                  : statusFilter !== 'All'
   580→                    ? `No companies with status "${statusFilter}".`
   581→                    : 'No companies yet.'}
   582→              </div>
   583→            )}
   584→          </CardContent>
   585→        </Card>
   586→      )}
   587→
   588→      {/* Pagination & info */}
   589→      {total > 0 && (
   590→        <div className="flex flex-col sm:flex-row items-center justify-between gap-4">
   591→          <div className="flex items-center gap-3">
   592→            <span className="text-sm text-stone-500">
   593→              Showing {showingFrom} to {showingTo} of {total} companies
   594→            </span>
   595→            <div className="flex items-center gap-1.5">
   596→              <span className="text-xs text-stone-500">Per page:</span>

   602→                  <SelectItem value="10">10</SelectItem>

   604→                  <SelectItem value="50">50</SelectItem>
   605→                </SelectContent>
   606→              </Select>
   607→            </div>
   608→          </div>
   609→          {totalPages > 1 && (
   610→            <Pagination>
   611→              <PaginationContent>
   612→                <PaginationItem>
   613→                  <PaginationPrevious
   614→                    onClick={() => setPage((p) => Math.max(1, p - 1))}
   615→                    className={page <= 1 ? 'pointer-events-none opacity-50' : 'cursor-pointer'}
   616→                  />
   617→                </PaginationItem>
   618→                {paginationRange[0] > 1 && (
   619→                  <>
   620→                    <PaginationItem>
   621→                      <PaginationLink onClick={() => setPage(1)} className="cursor-pointer">1</PaginationLink>
   622→                    </PaginationItem>
   623→                    {paginationRange[0] > 2 && <PaginationEllipsis />}
   624→                  </>
   625→                )}
   626→                {paginationRange.map((p) => (
   627→                  <PaginationItem key={p}>
   628→                    <PaginationLink
   629→                      isActive={p === page}
   630→                      onClick={() => setPage(p)}
   631→                      className="cursor-pointer"
   632→                    >
   633→                      {p}
   634→                    </PaginationLink>
   635→                  </PaginationItem>
   636→                ))}
   637→                {paginationRange[paginationRange.length - 1] < totalPages && (
   638→                  <>
   639→                    {paginationRange[paginationRange.length - 1] < totalPages - 1 && <PaginationEllipsis />}
   640→                    <PaginationItem>
   641→                      <PaginationLink
   642→                        onClick={() => setPage(totalPages)}
   643→                        className="cursor-pointer"
   644→                      >
   645→                        {totalPages}
   646→                      </PaginationLink>
   647→                    </PaginationItem>
   648→                  </>
   649→                )}
   650→                <PaginationItem>
   651→                  <PaginationNext
   652→                    onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   653→                    className={page >= totalPages ? 'pointer-events-none opacity-50' : 'cursor-pointer'}
   654→                  />
   655→                </PaginationItem>
   656→              </PaginationContent>
   657→            </Pagination>
   658→          )}
   659→        </div>
   660→      )}
   661→
   662→      {/* Create dialog */}
   663→      <CompanyFormDialog
   664→        open={creating}
   665→        onOpenChange={setCreating}
   666→        lookup={lookup}
   667→        customColumns={customColumns}
   668→        onSaved={() => { setCreating(false); load() }}
   669→      />
   670→      {/* Edit dialog */}
   671→      <CompanyFormDialog
   672→        open={!!editing}
   673→        onOpenChange={(o) => !o && setEditing(null)}
   674→        company={editing}
   675→        lookup={lookup}
   676→        customColumns={customColumns}
   677→        onSaved={() => { setEditing(null); load() }}
   678→      />
   679→      {/* Delete confirm */}
   680→      <AlertDialog open={!!deleting} onOpenChange={(o) => !o && setDeleting(null)}>
   681→        <AlertDialogContent>
   682→          <AlertDialogHeader>
   683→            <AlertDialogTitle>Delete company "{deleting?.name}"?</AlertDialogTitle>
   684→            <AlertDialogDescription>
   685→              This permanently removes the company, its founder attachments and all annual data. This cannot be undone.
   686→            </AlertDialogDescription>
   687→          </AlertDialogHeader>
   688→          <AlertDialogFooter>
   689→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   690→            <AlertDialogAction
   691→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   692→              onClick={async () => {
   693→                try {
   694→                  await api(`/api/admin/companies/${deleting!.id}`, { method: 'DELETE' })
   695→                  toast({ title: 'Deleted', description: `${deleting!.name} removed` })
   696→                  setDeleting(null)
   697→                  load()
   698→                } catch (e: any) {
   699→                  toast({ title: 'Error', description: e.message, variant: 'destructive' })
   700→                }
   701→              }}
   702→            >
   703→              Delete
   704→            </AlertDialogAction>
   705→          </AlertDialogFooter>
   706→        </AlertDialogContent>
   707→      </AlertDialog>
   708→    </div>
   709→  )
   710→}

   715→      <div className="text-xs text-muted-foreground">{label}</div>
